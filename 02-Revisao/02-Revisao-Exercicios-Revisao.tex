\begin{exercicios}{Revisão}

    \begin{exercicio}
        Determine o custo mínimo $C$ (em reais) de um certo produto, dado que
        \[5 (C - 25) \geq 1,75 + 2,5C\]	
    \end{exercicio}
    
    \begin{exercicio}
        Determine o lucro máximo $P$ (em reais) resultante de uma certa transação, dado que
        \[6 (P - 2500) \leq 4 (P + 2400)\]	
    \end{exercicio}
        
    \begin{exercicio}
        Determine as distâncias entre os pontos
        \[\left(\frac{1}{2}, \sqrt{3}\right) \text{ e } \left(- \frac{1}{2}, 2 \sqrt{3}\right)\] 				
    \end{exercicio}
    
    \begin{exercicio}
        Determine o custo mínimo $C$ (em reais) dado que
        \[2 (1,5C + 80) \leq 2 (2,5C - 20)\]
    \end{exercicio}
    
    \begin{exercicio}
        Encontre o retorno máximo $R$ (em reais) dado que
        \[12 (2R - 320) \leq 4 (3R + 240)\]
    \end{exercicio}
    
\end{exercicios}
