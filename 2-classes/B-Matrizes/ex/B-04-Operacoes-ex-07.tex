\begin{question}
Considerando
\(
  A =
  \begin{pmatrix}
    1 & -2 \\
    3 & 4
  \end{pmatrix}
\)
avalie $3A$
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
\begin{align*}
  \onslide<+->{3A}
  \onslide<+->{& = 3 \begin{pmatrix}
    1 & -2 \\
    3 & 4
  \end{pmatrix}
  \\[2mm]}
  \onslide<+->{
  & \setlength{\arraycolsep}{12pt}
  = \begin{pmatrix}
    3 \times 1 & 3(-2) \\
    3 \times 3 & 3 \times 4
  \end{pmatrix}
  \\[2mm]}
  \onslide<+->{& = \begin{pmatrix}
    3 & -6 \\
    9 & 12
  \end{pmatrix}}
\end{align*}
\end{resolution}
