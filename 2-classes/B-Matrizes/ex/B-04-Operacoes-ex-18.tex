\begin{question}
Considerando os vetores \;
\(
  u =
  \begin{pmatrix}
    2  \\
    -1 \\
    3
  \end{pmatrix}
\) \;
e
\; \(
  v =
  \begin{pmatrix}
    1  \\
    4  \\
    -2
  \end{pmatrix}
\),\;
calcule
\(
  uv^t
\)
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
\begin{align*}
  \onslide<+->{uv^t}
  \onslide<+->{& =
  \begin{pmatrix}
    2  \\
    -1 \\
    3
  \end{pmatrix}
  \begin{pmatrix}
    1  \\
    4  \\
    -2
  \end{pmatrix}^t}
  \onslide<+->{\;=\;
  \begin{pmatrix}
    2  \\
    -1 \\
    3
  \end{pmatrix}
  \begin{pmatrix}
    1 & 4 & -2
  \end{pmatrix}
  \\}
  \onslide<+->{& =
  \begin{pmatrix}
    2\times 1 & 2\times 4 & 2\times (-2) \\
    (-1)1     & (-1)4     & (-1)(-2)     \\
    3\times 1 & 3\times 4 & 3\times (-2)
  \end{pmatrix}
  \\}
  \onslide<+->{& =
  \begin{pmatrix}
    2  & 8  & -4 \\
    -1 & -4 & 2  \\
    3  & 12 & -6
  \end{pmatrix}}
\end{align*}
\end{resolution}
