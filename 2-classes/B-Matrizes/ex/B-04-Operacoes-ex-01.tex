\begin{question}
Determine $x$ e $y$ sabendo que
\[
  \begin{pmatrix}
    2x & 3   \\
    4  & y-1
  \end{pmatrix}
  =
  \begin{pmatrix}
    8 & 3 \\
    4 & 5
  \end{pmatrix}
\]
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
  \onslide<+->{Comparando os elementos correspondentes}
  \begin{columns}
  \column{0.4\linewidth}
    \begin{align*}
      \onslide<+->{2x & = 8           \\}
      \onslide<+->{x  & = \frac{8}{2} \\}
      \onslide<+->{x  & = 4}
    \end{align*}
  \column{0.4\linewidth}
    \begin{align*}
      \onslide<+->{y-1 & = 5     \\}
      \onslide<+->{y   & = 5 + 1 \\}
      \onslide<+->{y   & = 4}
    \end{align*}
  \end{columns}
\end{resolution}
