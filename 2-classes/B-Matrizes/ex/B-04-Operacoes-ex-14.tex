\begin{question}
Calcule $AJ$, para \;
\(
  A =
  \begin{pmatrix}
    1 & 2 & 3 \\
    4 & 5 & 6
  \end{pmatrix}
 \)
 \; e \;
 \(
  J =
  \begin{pmatrix}
    1 & 1 \\
    1 & 1 \\
    1 & 1
  \end{pmatrix}
\)
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
\begin{align*}
  AJ
  & =
  \begin{pmatrix}
    1 & 2 & 3 \\
    4 & 5 & 6
  \end{pmatrix}
  \begin{pmatrix}
    1 & 1 \\
    1 & 1 \\
    1 & 1
  \end{pmatrix}
  \\
  & \setlength{\arraycolsep}{12pt}
  = \begin{pmatrix}
    1\times1+2\times1+3\times1 & 1\times1+2\times1+3\times1 \\
    4\times1+5\times1+6\times1 & 4\times1+5\times1+6\times1
  \end{pmatrix}
  \\
  & =
  \begin{pmatrix}
    6  & 6  \\
    15 & 15
  \end{pmatrix}
\end{align*}
\end{resolution}
