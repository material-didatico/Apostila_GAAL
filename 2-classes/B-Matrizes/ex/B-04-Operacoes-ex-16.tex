%------------------------------------------------------------------------------%
\begin{question}
Determine a matriz transposta de
\(
  A =
  \begin{pmatrix}
    2  & -1 & 0 \\
    3  & 4  & 5 \\
    -2 & 1  & 6
  \end{pmatrix}
\)
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
  \[
    A^t
    \pause
    \;=\;
    \begin{pmatrix}
      2  & -1 & 0 \\
      3  & 4  & 5 \\
      -2 & 1  & 6
    \end{pmatrix}^t
    \pause
    \;=\;
    \begin{pmatrix}
      2  & 3 & -2 \\
      -1 & 4 & 1  \\
      0  & 5 & 6
    \end{pmatrix}
  \]
\end{resolution}
