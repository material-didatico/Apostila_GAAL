\begin{question}
  Determine $x$, $y$ e $z$ sabendo que
  \[
    \begin{pmatrix}
      x+1 & 2y  \\
      3   & z-2
    \end{pmatrix}
    =
    \begin{pmatrix}
      5 & 8 \\
      3 & 7
    \end{pmatrix}
  \]
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
  \onslide<+->{Comparando os elementos correspondentes}
  \begin{columns}
  \column{0.33\linewidth}
    \begin{align*}
      \onslide<+->{x+1 & = 5     \\}
      \onslide<+->{x   & = 5 - 1 \\}
      \onslide<+->{x   & = 4}
    \end{align*}
  \column{0.33\linewidth}
    \begin{align*}
      \onslide<+->{2y & = 8           \\}
      \onslide<+->{y  & = \frac{8}{2} \\}
      \onslide<+->{y  & = 4}
    \end{align*}
  \column{0.33\linewidth}
    \begin{align*}
      \onslide<+->{z-2 & = 7     \\}
      \onslide<+->{z   & = 7 + 2 \\}
      \onslide<+->{z   & = 9}
    \end{align*}
  \end{columns}
\end{resolution}
