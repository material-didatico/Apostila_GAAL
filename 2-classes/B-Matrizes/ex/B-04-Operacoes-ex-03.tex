\begin{question}
  Considere
  \quad
  \(
    A =
    \begin{pmatrix}
      1 & 2 \\
      3 & 4
    \end{pmatrix}
  \)
  \quad e \quad
  \(
    B =
    \begin{pmatrix}
      5 & -1 \\
      2 & 3
    \end{pmatrix}
  \)
  \quad
  avalie $A+B$
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
\begin{align*}
  \onslide<+->{A + B}
  \onslide<+->{& = \begin{pmatrix}
        1 & 2 \\
        3 & 4
      \end{pmatrix}
  +
      \begin{pmatrix}
        5 & -1 \\
        2 & 3
      \end{pmatrix}
  \\[2mm]}
  \onslide<+->{
  & \setlength{\arraycolsep}{12pt}
  = \begin{pmatrix}
    1+5 & 2+(-1) \\
    3+2 & 4+3
  \end{pmatrix}
  \\[2mm]}
  \onslide<+->{& =
  \begin{pmatrix}
    6 & 1 \\
    5 & 7
  \end{pmatrix}}
\end{align*}
\end{resolution}
