%------------------------------------------------------------------------------%
\begin{question}
Determine a transposta de
\(
  A =
  \begin{pmatrix}
    1 & 2 & 3 \\
    4 & 5 & 6
  \end{pmatrix}
\)
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
  \[
    A^t
    \pause
    \;=\;
    \begin{pmatrix}
      1 & 2 & 3 \\
      4 & 5 & 6
    \end{pmatrix}^t
    \pause
    \;=\;
    \begin{pmatrix}
      1 & 4 \\
      2 & 5 \\
      3 & 6
    \end{pmatrix}
  \]
\end{resolution}
