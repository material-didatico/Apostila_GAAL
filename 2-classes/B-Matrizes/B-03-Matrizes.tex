Nomenclatura
\begin{itemize}
  \item Cada sequência de elementos dispostos na horizontal é uma \textbf{linha} da matriz. \\
    \pause
    A primeira linha da matriz acima é $[80 \ 38 \ 24 \ 8 \ 6 \ 29]$.

    \pause

  \item Cada sequência de elementos dispostos na vertical é uma \textbf{coluna} da matriz. \\
    \pause
    A quarta coluna da matriz acima é $\left[ \begin{array}{c}8 \\
    10 \\
    6 \\
    12
    \end{array} \right]$
\end{itemize}
\end{frame}

%------------------------------------------------------------------------------%
\begin{frame}{Notação}
\begin{itemize}[<+->]
  \item Na notação, $A_{m \times n}$ \\
    $A$ é o nome da matriz \\
    $m$ e $n$ são os índices e indicam a ordem da matriz

  \item Elementos da matriz são denotados por $a_{ij}$
  \[
    a_{13} = 24\qquad
    a_{31} = 69\qquad
    a_{44} = 12\qquad
    a_{46} = 18
  \]
  \item A matriz pode ser denotada por
  \[
     A = [a_{ij}]_{m \times n}
  \]
\end{itemize}
\end{frame}

%------------------------------------------------------------------------------%
\begin{frame}{Matrizes Especiais}
\begin{itemize}[<+->]
  \item Uma matriz $A_{n \times n}$ é dita matriz quadrada
  \item Uma matriz $\mathcal{O} = [0]_{m\times n}$, que tem todos os elementos nulos, é uma matriz nula
  \item Uma matriz $B_{1\times n}$ é uma matriz coluna
  \item Uma matriz $C_{m\times 1}$ é uma matriz linha
  \item entre outros que veremos adiante
\end{itemize}
\end{frame}
