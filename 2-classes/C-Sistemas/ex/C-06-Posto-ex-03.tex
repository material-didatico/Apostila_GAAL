%------------------------------------------------------------------------------%
\begin{question}
  Determine o número de variáveis livres do sistema linear associado à matriz
  \[
    A =
    \begin{pmatrix}
      1 & 2 & 0 & -1 & 3 \\
      0 & 0 & 1 & 4  & 2 \\
      0 & 0 & 0 & 0  & 0
    \end{pmatrix}
  \]
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
  \[
    A =
    \begin{pmatrix}
      \emph{1} & 2 & 0        & -1 & 3 \\
      0        & 0 & \emph{1} & 4  & 2 \\
      0        & 0 & 0        & 0  & 0
    \end{pmatrix}
  \]
  \pause
  \[
    \posto(A) = 2
  \]

  \pause
  $A$ possui 5 colunas,
  \pause
  o sistema linear tem \emph{$n = 5$} incógnitas
  \[
    \pause
    \text{Variáveis livres}
    \pause
    \;=\;
    \posto(A) - n
    \pause
    \;=\;
    5 - 2
    \pause
    \;=\;
    3
  \]
\end{resolution}
