\begin{question}
Determine quais das matrizes abaixo estão em forma escalonada
\[
  A =
  \begin{bmatrix}
    1 & 2 & 3 \\
    0 & 1 & 4 \\
    0 & 0 & 2
  \end{bmatrix}
  \qquad
  B =
  \begin{bmatrix}
    1 & 2 & 3 \\
    0 & 0 & 4 \\
    0 & 2 & 5
  \end{bmatrix}
  \qquad
  C =
  \begin{bmatrix}
    0 & 1 & 3 \\
    0 & 0 & 2 \\
    0 & 0 & 0
  \end{bmatrix}
\]
\end{question}

%------------------------------------------------------------------------------%
\begin{resolutiont}{Solução}
\onslide<+->{\[
  A = \begin{bmatrix}
    1 & 2 & 3 \\
    0 & 1 & 4 \\
    0 & 0 & 2
  \end{bmatrix}
\]}
\onslide<+->{A matriz $A$ está escalonada}
\begin{enumerate}[<+->]
  \item os pivôs estão progressivamente mais à direita
  \item todos os elementos abaixo dos pivôs são nulos
\end{enumerate}
\end{resolutiont}

%------------------------------------------------------------------------------%
\begin{resolutiont}{Solução}
\onslide<+->{\[
  B = \begin{bmatrix}
    1 & 2                   & 3                   \\
    0 & 0                   & {\color<3->{red} 4} \\
    0 & {\color<3->{red} 2} & 5
  \end{bmatrix}
\]}%
\onslide<+->{A matriz $B$ não está escalonada}  \\
\onslide<+->{o pivô da terceira linha está à esquerda do pivô da segunda linha}

\bigskip
\onslide<+->{Pode ser escalonada com a troca da segunda e terceira linhas}
\end{resolutiont}

%------------------------------------------------------------------------------%
\begin{resolutiont}{Solução}
\onslide<+->{\[
  C = \begin{bmatrix}
    0 & 1 & 3 \\
    0 & 0 & 2 \\
    0 & 0 & 0
  \end{bmatrix}
\]}
\onslide<+->{A matriz $C$ está escalonada}
\begin{enumerate}[<+->]
  \item os pivôs estão progressivamente mais à direita
  \item todos os elementos abaixo dos pivôs são nulos
  \item as linhas nulas estão abaixo das linhas não nulas
\end{enumerate}
\end{resolutiont}
