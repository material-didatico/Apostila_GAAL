\begin{question}
Transforme a matriz
\[
  A =
  \begin{bmatrix}
    1 & 2 & 1 \\
    2 & 5 & 4 \\
    1 & 2 & 3
  \end{bmatrix}
\]
em sua forma escalonada reduzida
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Escalonamento}
\begin{columns}
\column{0.3\linewidth}
  \[\;\;\;
  \begin{bmatrix}
    1 & 2 & 1 \\
    2 & 5 & 4 \\
    1 & 2 & 3
  \end{bmatrix}
  \]
  \pause
\column{0.7\linewidth}
  Eliminamos os elementos abaixo do primeiro pivô
  \\ \pause
  \(
    L_2 \leftarrow L_2 - 2 L_1
  \)
  \\ \pause
  \(
    L_3 \leftarrow L_3 - L_1
  \)
\end{columns}
\vspace{-\baselineskip}
\[
  \pause
  L'_2
  \;=\;
  L_2 - 2 L_1
  \pause
  \;=\;
  \begin{bmatrix}
    2 & 5 & 4
  \end{bmatrix}
  - 2
  \begin{bmatrix}
    1 & 2 & 1
  \end{bmatrix}
  \pause
  \;=\;
  \begin{bmatrix}
    0 & 1 & 2
  \end{bmatrix}
\]
\[
  \pause
  L'_3
  \;=\;
  L_3 - L_1
  \pause
  \;=\;
  \begin{bmatrix}
    1 & 2 & 3
  \end{bmatrix}
  -
  \begin{bmatrix}
    1 & 2 & 1
  \end{bmatrix}
  \pause
  \;=\;
  \begin{bmatrix}
    0 & 0 & 2
  \end{bmatrix}
\]
\pause
\[
  \begin{bmatrix}
    1 & 2 & 1 \\
    0 & 1 & 2 \\
    0 & 0 & 2
  \end{bmatrix}
  \qquad\qquad
  \pause
  \text{Matriz escalonada}
\]
\end{resolution}
