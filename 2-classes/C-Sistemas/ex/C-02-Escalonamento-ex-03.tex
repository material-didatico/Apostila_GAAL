\begin{question}
  Operação elementar: \emph{combinação de linhas}

  Considerando
  \(
    A =
    \begin{bmatrix}
      1 & 2 & 3 \\
      2 & 5 & 7
    \end{bmatrix}
  \)

  Produza um zero na linha 2 coluna 1, usando
  \[
    L_2\leftarrow L_2-2L_1
  \]
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
\[
  \begin{bmatrix}
    1 & 2 & 3 \\
    2 & 5 & 7
  \end{bmatrix}
\]

\pause
\[
  L'_2
  \pause
  \;=\;
   L_2 - 2L_1
   \pause
  \;=\;
  \begin{bmatrix}
    2 & 5 & 7
  \end{bmatrix}
  -
  2\begin{bmatrix}
    1 & 2 & 3
  \end{bmatrix}
  \pause
  \;=\;
  \begin{bmatrix}
    0 & 1 & 1
  \end{bmatrix}
\]

\pause
\[
  \begin{bmatrix}
    1 & 2 & 3 \\
    0 & 1 & 1
  \end{bmatrix}
\]
\end{resolution}
