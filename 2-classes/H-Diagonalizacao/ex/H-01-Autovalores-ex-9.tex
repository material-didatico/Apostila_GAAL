%------------------------------------------------------------------------------%
\begin{question}
  Determine os autovalores, os autovetores e os autoespaços da matriz
  \[
    A =
    \begin{pmatrix}
      1 & 1 & 0  \\
      0 & 3 & 0  \\
      0 & 0 & -2
    \end{pmatrix}
  \]
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Autovalores}
  \onslide<+->{Os autovalores são as raízes do polinômio característico}
  \begin{align*}
      \onslide<+->{p(\lambda)}
      \onslide<+->{& = \det(A-\lambda I)}
      \\
      \onslide<+->{& = \begin{vmatrix}
        1-\lambda & 1         & 0          \\
        0         & 3-\lambda & 0          \\
        0         & 0         & -2-\lambda
      \end{vmatrix}}
      \\
      \onslide<+->{& = (1-\lambda)(3-\lambda)(-2-\lambda)}
  \end{align*}

  \onslide<+->{Portanto
  \[
      \lambda_{1} = 1
      \qquad
      \lambda_{2} = 3
      \qquad
      \lambda_{3} = -2
  \]}
\end{resolution}

%------------------------------------------------------------------------------%
\begin{resolution}{Autovetores para $\lambda=1$}
\begin{columns}
\column{0.7\textwidth}
  \begin{align*}
      \onslide<+->{(A - 1I) \vec{x} & = \vec{0}}
      \\[3mm]
      \onslide<+->{\left(
        \begin{bmatrix}
          1 & 1 & 0  \\
          0 & 3 & 0  \\
          0 & 0 & -2
        \end{bmatrix}
      - \begin{bmatrix}
          1 & 0 & 0  \\
          0 & 1 & 0  \\
          0 & 0 & 1
        \end{bmatrix}
      \right)
      \vetortri{x}{y}{z}
      & = \vetortri{0}{0}{0}}
      \\[3mm]
      \onslide<+->{\begin{pmatrix}
        0 & 1 & 0  \\
        0 & 2 & 0  \\
        0 & 0 & -3
      \end{pmatrix}
      \vetortri{x}{y}{z}
      & = \vetortri{0}{0}{0}}
  \end{align*}

\column{0.3\textwidth}
  \begin{align*}
    \onslide<+->{x & \in \R \\}
    \onslide<+->{y & = 0    \\}
    \onslide<+->{z & = 0}
  \end{align*}

  \onslide<+->{Autovetor
  \[
    \vec{v}_1 = \vetortri{1}{0}{0}
  \]}
\end{columns}
\end{resolution}

%------------------------------------------------------------------------------%
\begin{resolution}{Autovetores para $\lambda=3$}
\begin{columns}
\column{0.7\textwidth}
  \begin{align*}
    \onslide<+->{(A - 3I) \vec{x} & = \vec{0}}
    \\[3mm]
    \onslide<+->{\left(
      \begin{bmatrix}
        1 & 1 & 0  \\
        0 & 3 & 0  \\
        0 & 0 & -2
      \end{bmatrix}
    - \begin{bmatrix}
        3 & 0 & 0  \\
        0 & 3 & 0  \\
        0 & 0 & 3
      \end{bmatrix}
    \right)
    \vetortri{x}{y}{z}
    & = \vetortri{0}{0}{0}}
    \\[3mm]
    \onslide<+->{\begin{pmatrix}
      -2 & 1 & 0  \\
      0  & 0 & 0  \\
      0  & 0 & -5
    \end{pmatrix}
    \vetortri{x}{y}{z}
    & = \vetortri{0}{0}{0}}
  \end{align*}

\column{0.3\textwidth}
  \begin{align*}
    \onslide<+->{z     & = 0 \\[4mm]}
    \onslide<+->{-2x+y & = 0 \\}
    \onslide<+->{y     & = 2x  }
  \end{align*}

  \onslide<+->{Autovetor
  \[
    \vec{v}_2 = \vetortri{1}{2}{0}
  \]}
\end{columns}
\end{resolution}

%------------------------------------------------------------------------------%
\begin{resolution}{Autovetores para $\lambda=-2$}
\begin{columns}
\column{0.7\textwidth}
  \begin{align*}
    \onslide<+->{(A-(-2)I)\vec{x}&=\vec{0}}
    \\[3mm]
    \onslide<+->{\left(
      \begin{bmatrix}
        1 & 1 & 0  \\
        0 & 3 & 0  \\
        0 & 0 & -2
      \end{bmatrix}
    + \begin{bmatrix}
        2 & 0 & 0  \\
        0 & 2 & 0  \\
        0 & 0 & 2
      \end{bmatrix}
    \right)
    \vetortri{x}{y}{z}
    & = \vetortri{0}{0}{0}}
    \\[3mm]
    \onslide<+->{\begin{pmatrix}
        3 & 1 & 0\\
        0 & 5 & 0\\
        0 & 0 & 0
    \end{pmatrix}
    \vetortri{x}{y}{z}
    & = \vetortri{0}{0}{0}}
  \end{align*}

\column{0.3\textwidth}
  \begin{align*}
    \onslide<+->{     y & = 0 \\[3mm]}
    \onslide<+->{3x + y & = 0 \\}
    \onslide<+->{     x & = 0 \\[3mm]}
    \onslide<+->{z      & \in\R}
  \end{align*}

  \onslide<+->{Autovetor
  \[
    \vec{v}_3 = \vetortri{0}{0}{1}
  \]}
\end{columns}
\end{resolution}

%------------------------------------------------------------------------------%
\begin{resolution}{Resposta}
  \begin{columns}
  \column{0.33\textwidth}
  \[
    \lambda_{1} = 1
  \]
  \[
    \vec{v}_1 = \vetortri{1}{0}{0}
  \]
  \[
    E(2) = \ger\left\{\vec{v}_1\right\}
  \]
  \column{0.33\textwidth}
  \[
    \lambda_{2} = 3
  \]
  \[
    \vec{v}_2 = \vetortri{1}{2}{0}
  \]
  \[
    E(3) = \ger\left\{\vec{v}_2\right\}
  \]
  \column{0.35\textwidth}
  \[
    \lambda_{3} = -2
  \]
  \[
    \vec{v}_3 = \vetortri{0}{0}{1}
  \]
  \[
    E(-2) = \ger\left\{\vec{v}_3\right\}
  \]
  \end{columns}
\end{resolution}
