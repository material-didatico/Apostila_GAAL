\section{Dependência e Independência Linear}

%------------------------------------------------------------------------------%
\begin{frame}{Combinação Linear}
  Sejam
  \[
    \vec{v}_1,\; \vec{v}_2,\; \ldots,\; \vec{v}_k \in \R^n
  \]
  \[
    \pause
    \alpha_1,\, \alpha_2,\, \ldots,\, \alpha_k \in \R
  \]

  \pause
  Uma \emph{combinação linear} é
  \[
    \pause
    \alpha_1\vec{v}_1 \;+\;
    \alpha_2\vec{v}_2 \;+\;
    \cdots            \;+\;
    \alpha_k\vec{v}_k
  \]
\end{frame}

%------------------------------------------------------------------------------%
\begin{frame}{Independência Linear}
  Os vetores
  \pause
  \(
    \vec{v}_1, \ldots, \vec{v}_k
  \)
  \pause
  são \emph{Linearmente Independentes} (LI) se
  \[
    \pause
    \alpha_1\vec{v}_1 + \cdots + \alpha_k\vec{v}_k = \vec{0}
  \]
  \pause
  só é possível com
  \[
    \pause
    \alpha_1 = \cdots = \alpha_k = 0
  \]

  \pause
  \medskip
  Não é possível escrever nenhum vetor como combinação dos demais
\end{frame}

%------------------------------------------------------------------------------%
\begin{frame}{Dependência Linear}
  Os vetores
  \[
    \pause
    \vec{v}_1, \ldots, \vec{v}_k
  \]
  \pause
  são \emph{Linearmente Dependentes} (LD), se existem escalares
  \[
    \pause
    \alpha_1, \ldots, \alpha_k
  \]
  \pause
  \emph{não todos nulos} tais que
  \[
    \pause
    \alpha_1\vec{v}_1 + \cdots + \alpha_k\vec{v}_k
    \pause
    = \vec{0}
  \]
\end{frame}

%------------------------------------------------------------------------------%
\begin{frame}{Dependência Linear}
  \onslide<+->{Assumindo que \(\alpha_1 \neq 0\)}

  \onslide<+->{Podemos escrever \(\vec{v}_1\) como combinação dos demais}
  \begin{align*}
    \onslide<+->{\alpha_1\vec{v}_1 & + \alpha_2\vec{v}_2 + \cdots + \alpha_k\vec{v}_k = \vec{0} \\}
    \onslide<+->{\alpha_1\vec{v}_1 & = - \alpha_2\vec{v}_2 - \cdots - \alpha_k\vec{v}_k \\}
    \onslide<+->{\vec{v}_1 & = - \frac{\alpha_2}{\alpha_1}\vec{v}_2 - \cdots - \frac{\alpha_k}{\alpha_1}\vec{v}_k }
  \end{align*}
\end{frame}

\section{Matrizes e Sistema Lineares}

%------------------------------------------------------------------------------%
\begin{frame}{Sistema Homogêneo}
  Colocando os vetores como colunas de uma matriz
  \[
    \pause
    A =
    \begin{bmatrix}
      \vec{v}_1 & \vec{v}_2 & \cdots & \vec{v}_k
    \end{bmatrix}
  \]
  \pause
  A condição
  \[
    \alpha_1\vec{v}_1 + \cdots + \alpha_k\vec{v}_k = \vec{0}
  \]
  \pause
  Corresponde à
  \[
    \pause
    \begin{bmatrix}[c]
      |         & |         &        & |         \\
      \vec{v}_1 & \vec{v}_2 & \cdots & \vec{v}_k \\
      |         & |         &        & |
    \end{bmatrix} \;
    \begin{bmatrix}[c]
      \alpha_1 \\
      \vdots   \\
      \alpha_k
    \end{bmatrix}
    \pause
    \;=\; A \vec{x}
    \pause
    \;=\; \vec{0}
  \]
\end{frame}

%------------------------------------------------------------------------------%
\begin{frame}{Critério para Independência Linear}
  Os vetores são linearmente independentes
  \pause
  se o sistema
  \[
    \pause
    A\vec{x} = \vec{0}
  \]
  \pause
  possui somente a solução trivial
  \(
    \pause
    \vec{x} = \vec{0}
  \)
\end{frame}

%------------------------------------------------------------------------------%
\begin{frame}{Determinante}
  Dados $n$ vetores
  \[
    \pause
    \vec{v}_1, \vec{v}_2, \ldots, \vec{v}_n \in \R^n
  \]
  \pause
  formamos a matriz quadrada
  \[
    \pause
    A =
    \begin{bmatrix}
      \vec{v}_1 & \vec{v}_2 & \cdots & \vec{v}_n
    \end{bmatrix}
  \]
  \pause
  Se
  \[
    \det(A) \neq 0
  \]
  \pause
  os vetores são linearmente independentes
\end{frame}

%------------------------------------------------------------------------------%
\begin{frame}{Vetor nulo}
  Qualquer conjunto que contenha
  \[
    \vec{0}
  \]
  \pause
  é linearmente dependente
\end{frame}

%------------------------------------------------------------------------------%
\begin{frame}{Múltiplos}
  Se
  \[
    \vec{v}_2 = \alpha\vec{v}_1
  \]
  \pause
  para algum
  \[
    \alpha\in\R
  \]
  \pause
  então
  \[
    \vec{v}_1
    \quad\text{e}\quad
    \vec{v}_2
  \]
  \pause
  são linearmente dependentes
\end{frame}
