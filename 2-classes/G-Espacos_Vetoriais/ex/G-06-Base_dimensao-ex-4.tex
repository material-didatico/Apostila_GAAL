\begin{question}
  Determine se
  \[
    \mathcal{S} =\left\{
      \vetor{1}{0},
      \vetor{0}{1},
      \vetor{1}{1}
    \right\}
  \]
  é uma base de $\R^{2}$
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
  O conjunto gera $\R^{2}$ pois
  \[
    \vetor{x}{y} = x\vetor{1}{0} + y\vetor{0}{1}
  \]

  \pause
  Porém
  \[
    \vetor{1}{1} = \vetor{1}{0} + \vetor{0}{1}
  \]
  \pause
  Logo os vetores são LD

  \pause
  Portanto $\mathcal{S}$ não é uma base
\end{resolution}
