%------------------------------------------------------------------------------%
\begin{question}
  Determine a forma matricial \(T(v)=Av\) da transformação linear
  \[
    T(x, y) = \vetor{2x-y}{x+3y}
  \]
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
  Precisamos determinar uma matriz $A$
  \pause
  tal que
  \[
    A\vetor{x}{y} = \vetor{2x-y}{x+3y}
  \]

  \pause
  Como
  \[
    \begin{pmatrix}
      2 & -1 \\
      1 & 3
    \end{pmatrix}
    \begin{pmatrix}
      x \\
      y
    \end{pmatrix}
    \;=\;
    \begin{pmatrix}
      2x-y \\
      x+3y
    \end{pmatrix}
  \]

  \pause
  \[
    A = \begin{pmatrix}
          2 & -1 \\
          1 & 3
        \end{pmatrix}
  \]
\end{resolution}

\endinput
