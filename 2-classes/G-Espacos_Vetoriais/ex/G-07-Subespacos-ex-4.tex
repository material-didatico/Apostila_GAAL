\begin{question}
  Determine se
  \[
    W = \left\{ \vetortri{x}{y}{z} \in\R^{3} \st x+y-z = 2 \right\}
  \]
  é um subespaço de $\R^{3}$
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Vetor Nulo}
  O vetor nulo \(\vetortri{0}{0}{0}\) pertence a $W$?

  \pause
  \[
    x + y - z
    \pause
    = 0 + 0 - 0
    \pause
    = 0
    \pause
    \neq 2
  \]

  \pause
  Portanto
  \[
    \vec{0} \notin W
  \]

  \pause
  Portanto, $W$ não é subespaço de $\R^{2}$
\end{resolution}
