\begin{question}
  Determine se
  \[
  W=
  \left\{
  f:\R\to\R
  \;\middle|\;
  f(0)=0
  \right\}
  \]
  é um subespaço do espaço de funções reais
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
  A função nula satisfaz
  \begin{align*}
  0(0)&=0
  \end{align*}
  Logo
  \[
  0\in W
  \]
  Sejam $f,g\in W$.
  Então
  \[
  f(0)=0
  \qquad
  g(0)=0
  \]
  Para $\lambda,\mu\in\R$
  \begin{align*}
  (\lambda f+\mu g)(0)
  &=\lambda f(0)+\mu g(0)\\
  &=\lambda\cdot0+\mu\cdot0\\
  &=0
  \end{align*}
  Logo
  \[
  \lambda f+\mu g\in W
  \]
  Portanto $W$ é subespaço.

\end{resolution}
