\begin{question}
  Determine se
  \[
    W = \left\{ \vetor{x}{y} \in\R^{2} \st y=2x+1 \right\}
  \]
  é um subespaço de $\R^{2}$
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Vetor Nulo}
  O vetor nulo \(\vetor{0}{0}\) pertence a $W$?

  \pause
  Se $x=0$ então
  \[
    \pause
    y
    \pause
    = 3x + 1
    \pause
    = 3 \times 0 + 1
    \pause
    = 1
    \pause
    \neq 0
  \]

  \pause
  Logo
  \(
    \vec{0} \notin W
  \)

  \pause
  Portanto, $W$ não é subespaço de $\R^{2}$
\end{resolution}
