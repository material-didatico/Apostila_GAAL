\begin{question}
  Considere a base ortogonal
  \[
    B=\{(1,1,0),(1,-1,0),(0,0,2)\}
  \]
  Determine as coordenadas de
  \[
    \vec{x}=(3,-1,4)
  \]
  na base $B$
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
  \onslide<+->{Se
  \(
    \vec{x} = c_1\vec{v}_1 + c_2\vec{v}_2+c_3\vec{v}_3
  \)}
\begin{columns}
\column{0.25\textwidth}
  \begin{align*}
    \onslide<+->{c_1 & = \frac{\vec{x}\cdot\vec{v}_1}{\norm{\vec{v}_1}^{2}} \\}
    \onslide<+->{    & = \frac{3-1}{2}                                      \\}
    \onslide<+->{    & = 1}
  \end{align*}

\column{0.25\textwidth}
  \begin{align*}
    \onslide<+->{c_2 & = \frac{\vec{x}\cdot\vec{v}_2} {\norm{\vec{v}_2}^{2}} \\}
    \onslide<+->{    & = \frac{3+1}{2}                                       \\}
    \onslide<+->{    & =2}
  \end{align*}
\column{0.3\textwidth}
  \begin{align*}
    \onslide<+->{c_3 & = \frac{\vec{x}\cdot\vec{v}_3}{\norm{\vec{v}_3}^{2}} \\}
    \onslide<+->{    & = \frac{8}{4}                                        \\}
    \onslide<+->{    & = 2}
  \end{align*}
\end{columns}
\end{resolution}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
  \[
    [\vec{x}]_B=
    \begin{bmatrix}
      1 \\
      2 \\
      2
    \end{bmatrix}
  \]
\end{resolution}
