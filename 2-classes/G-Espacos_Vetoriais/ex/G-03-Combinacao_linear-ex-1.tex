\begin{question}
  Dados
  \[
    \vec{v} = (1, 2)
    \qquad
    \vec{w} = (2, -1)
  \]
  Determine a combinação linear
  \[
    3\vec{v} - 2\vec{w}
  \]
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
  \begin{align*}
    \onslide<+->{3\vec{v}-2\vec{w}}
    \onslide<+->{& = 3(1, 2)-2(2, -1) \\}
    \onslide<+->{& = (3, 6)-(4, -2)   \\}
    \onslide<+->{& = (-1, 8)}
  \end{align*}
\end{resolution}
