\begin{frame}{Combinação Linear de Dois Vetores}
  Dados os vetores
  \(
    \vec{v}
  \) e
  \(
    \vec{w}
  \)

  e os escalares
  \(
    a
  \) e
  \(
    b
  \)

  uma combinação linear é um vetor da forma
  \[
    a \vec{v} + b \vec{w}
  \]
\end{frame}

%------------------------------------------------------------------------------%
\begin{frame}{Combinação Linear}
  Dados vetores
  \(
    \vec{v}_1, \ldots, \vec{v}_k
  \)

  e escalares
  \(
    c_1, \ldots, c_k
  \)

  uma combinação linear é um vetor da forma
  \[
    c_1\vec{v}_1 + \cdots + c_k\vec{v}_k
  \]
\end{frame}
