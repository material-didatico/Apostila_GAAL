\begin{frame}{Três vetores determinam um paralelepípedo}
  Três vetores
  \(
    \vec{v}, \vec{w}, \vec{z}
  \)
  com a mesma origem

  \pause
  \bigskip
  Determinam um paralelepípedo no espaço $\R^3$

  \pause
  \bigskip
  O volume desse paralelepípedo é
  \[
    V \;=\;
    \abs{\left( \vec{v}\times\vec{w} \right)\cdot\vec{z}}
  \]

  \pause
  \bigskip
  O valor absoluto é necessário porque volume é uma grandeza não negativa
\end{frame}

\section{Exemplos}

\begin{question}
  Determine o volume do paralelepípedo determinado pelos vetores
  \[
    \vec{v} = (1, 2, 0)
  \]
  \[
    \vec{w} = (2, -1, 1)
  \]
  \[
    \vec{z} = (1, 0, 3)
  \]
\end{question}

\begin{resolution}{Exemplo 1 — Resolução}
  \begin{align*}
    \onslide<+->{V & = \abs{\left( \vec{v}\times\vec{w} \right)\cdot\vec{z}}}
    \\
    \onslide<+->{& = \abs{\;\det \begin{pmatrix}
      1 & 2  & 0 \\
      2 & -1 & 1 \\
      1 & 0  & 3
    \end{pmatrix}\;}}
    \\
    \onslide<+->{& = \abs{\; \begin{vmatrix}
        -1 & 1 \\
        0  & 3
      \end{vmatrix}
      - 2 \begin{vmatrix}
        2 & 1 \\
        1 & 3
      \end{vmatrix}^{\vphantom{I}}\;}}
    \\
    \onslide<+->{& = \abs{-3-10} \\}
    \onslide<+->{& = 13}
  \end{align*}
\end{resolution}

%------------------------------------------------

\begin{question}
  Calcule o volume do paralelepípedo determinado pelos vetores
  \[
    \vec{v} = (2, 1, 1)
  \]
  \[
    \vec{w} = (1, -1, 2)
  \]
  \[
    \vec{z} =(3, 2, -1)
  \]
\end{question}

\begin{resolution}{Resolução}
  \begin{align*}
    \onslide<+->{p & = \left( \vec{v}\times\vec{w} \right)\cdot\vec{z}}
    \onslide<+->{\;=\; \begin{vmatrix}
                         2 & 1  & 1  \\
                         1 & -1 & 2  \\
                         3 & 2  & -1
                       \end{vmatrix}}
    \\
    \onslide<+->{& = 2
      \begin{vmatrix}
      -1&2\\
      2&-1
      \end{vmatrix}
      -
      \begin{vmatrix}
      1&2\\
      3&-1
      \end{vmatrix}
      +
      \begin{vmatrix}
      1&-1\\
      3&2
      \end{vmatrix}}
    \\
    \onslide<+->{& = 2(-3)-(-7)+5} \\
    \onslide<+->{& = 6}
  \end{align*}

  \[
    \onslide<+->{V}
    \onslide<+->{\;=\; \abs{6}}
    \onslide<+->{\;=\; 6}
  \]
\end{resolution}
