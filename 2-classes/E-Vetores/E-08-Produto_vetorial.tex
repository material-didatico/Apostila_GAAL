\section{Introdução}

%------------------------------------------------------------------------------%
\begin{frame}{Produto Vetorial}{}
  \onslide<1->{\emph{\color<2>{red}Apenas para vetores em $\R^3$}}

  \bigskip
  \onslide<3->{é possível definir uma operação de produto entre vetores}

  \bigskip
  \onslide<4->{que produz um novo vetor ortogonal a ambos os vetores originais}
\end{frame}

%------------------------------------------------------------------------------%
\begin{frame}{Produto Vetorial}
  Dados dois vetores não nulos
  \[
    \vec{v},\; \vec{w} \in \emph{\R^3}
  \]
  \pause
  O \emph{produto escalar} produz um número
  \[
    \vec{v} \cdot \vec{w} \in \emph{\R}
  \]
  \pause
  O \emph{produto vetorial} produz um vetor ortogonal aos dois vetores
  \[
    \vec{v} \times \vec{w} \in \emph{\R^3}
  \]
  \pause
  \emph{Não misture as notações: \; $\cdot$ \; $\times$}
\end{frame}

%------------------------------------------------------------------------------%
\begin{frame}{Definição Geométrica}
  Considere dois vetores com a mesma origem
  \[
    \vec{v},\; \vec{w} \in \R^3
  \]
  \pause
  O produto vetorial \(\vec{v}\times\vec{w}\)
  \pause
  \begin{itemize}[<+->]
      \item é ortogonal a $\vec{v}$
      \item é ortogonal a $\vec{w}$
      \item seu sentido é determinado pela regra da mão direita
  \end{itemize}
\end{frame}

%------------------------------------------------------------------------------%
\begin{frame}{Regra da Mão Direita}
  \onslide<+->{O sentido do produto vetorial é determinado pela regra da mão direita}
  \only<+->{\addgraphic{6}{9,2.5}{Right_hand_rule_cross_product}}

  \bigskip
  \begin{itemize}[<+->]
      \item aponte o dedo indicador na direção de $\vec{v}$
      \item aponte o dedo médio na direção a $\vec{w}$
      \item o polegar indica a direção de $\vec{v}\times\vec{w}$
  \end{itemize}

  \bigskip
  \onslide<+->{\emph{A ordem dos vetores é importante}}
  \onslide<+->{
  \[
    \vec{v}\times\vec{w} \;=\; -\left(\vec{w}\times\vec{v}\right)
  \]}
\end{frame}

\section{Definição}

%------------------------------------------------------------------------------%
\begin{frame}{Definição Algébrica}
  Dados \quad
  \(
    \vec{v} = \vetortri{v_1}{v_2}{v_3}
    \qquad
    \vec{w} = \vetortri{w_1}{w_2}{w_3}
  \)

  \pause
  \bigskip
  O produto vetorial é definido por
  \[
    \vec{v}\times\vec{w}
    = \vetortri{v_2w_3-v_3w_2}{v_3w_1-v_1w_3}{v_1w_2-v_2w_1}
  \]
\end{frame}

%------------------------------------------------------------------------------%
\begin{frame}{Representação Alternativa}
  Definindo os vetores
  \(
    \quad
    \vec{i} = \vetortri{1}{0}{0}
    \qquad
    \vec{j} = \vetortri{0}{1}{0}
    \qquad
    \vec{k} = \vetortri{0}{0}{1}
  \)

  \pause
  \bigskip
  Temos
   \[
    \vec{v}\times\vec{w}
    \;=\; \left( v_2 w_3 - v_3 w_2 \right) \vec{i}
    \;+\; \left( v_3 w_1 - v_1 w_3 \right) \vec{j}
    \;+\; \left( v_1 w_2 - v_2 w_1 \right) \vec{k}
  \]
\end{frame}

%------------------------------------------------------------------------------%
\begin{frame}{Forma por Determinante}
  Cálculo pelo determinante
  \[
    \vec{v}\times\vec{w}
    \;=\;
    \begin{vmatrix}[c]
      \vec{i} & \vec{j} & \vec{k} \\
      v_1     & v_2     & v_3     \\
      w_1     & w_2     & w_3
    \end{vmatrix}
  \]

  \pause
  Expandindo pela primeira linha
  \[
    \vec{v} \times \vec{w}
    \;=\;
    \begin{vmatrix}
      v_2 & v_3 \\
      w_2 & w_3
    \end{vmatrix}
    \vec{i}
    \;-\;
    \begin{vmatrix}
      v_1 & v_3 \\
      w_1 & w_3
    \end{vmatrix}
    \vec{j}
    \;+\;
    \begin{vmatrix}
      v_1 & v_2 \\
      w_1 & w_2
    \end{vmatrix}
    \vec{k}
  \]
  \pause
  Lembre-se do Teorema de Laplace -- \emph{Atenção aos sinais}
\end{frame}

\section{Norma do Produto Vetorial}

%------------------------------------------------------------------------------%
\begin{frame}{Norma do Produto Vetorial}
  Dados
  \[
    \vec{v} = \vetortri{v_1}{v_2}{v_3}
    \qquad
    \vec{w} = \vetortri{w_1}{w_2}{w_3}
  \]
  \pause
  Temos
  \[
    \vec{v}\times\vec{w}
    =
    \vetortri{v_2w_3-v_3w_2}{v_3w_1-v_1w_3}{v_1w_2-v_2w_1}
  \]
\end{frame}

%------------------------------------------------------------------------------%
\begin{frame}{Norma do Produto Vetorial}
  \begin{align*}
    \onslide<+->{\norm{\vec{v}\times\vec{w}}}
    \onslide<+->{& = \sqrt{ (v_2w_3-v_3w_2)^2 + (v_3w_1-v_1w_3)^2 + (v_1w_2-v_2w_1)^2 }  \\[5mm]}
    \onslide<+->{& \;\; \vdots \qquad \text{manipulações simples, mas longas e tediosas} \\[5mm]}
    \onslide<+->{& = \sqrt{\norm{\vec{v}}^2 \, \norm{\vec{w}}^2 - \left(\vec{v}\cdot\vec{w}\right)^2}}
  \end{align*}
\end{frame}

%------------------------------------------------------------------------------%
\begin{frame}{Norma do Produto Vetorial}
  \begin{align*}
    \onslide<+->{\norm{\vec{v}\times\vec{w}}
                 & = \sqrt{\norm{\vec{v}}^2 \, \norm{\vec{w}}^2 - \left(\vec{v}\cdot\vec{w}\right)^2} \\}
    \onslide<+->{& = \sqrt{\norm{\vec{v}}^2 \, \norm{\vec{w}}^2 - \left(\norm{\vec{v}}\, \norm{\vec{w}} \cos(\theta)\right)^2} \\}
    \onslide<+->{& = \sqrt{\norm{\vec{v}}^2 \, \norm{\vec{w}}^2 - \norm{\vec{v}}^2 \norm{\vec{w}}^2 \cos^2(\theta)} \\}
    \onslide<+->{& = \sqrt{\norm{\vec{v}}^2 \, \norm{\vec{w}}^2 \left( 1 -  \cos^2(\theta)\right)} \\}
    \onslide<+->{& = \sqrt{\norm{\vec{v}}^2 \, \norm{\vec{w}}^2 \sin^2(\theta)} }
  \end{align*}
  \onslide<+->{Para \(0 \leq \theta \leq \pi\) temos \(\sin(\theta) \geq 0\)}
  \onslide<+->{\[
    \norm{\vec{v}\times\vec{w}} \;=\; \norm{\vec{v}} \, \norm{\vec{w}} \sin(\theta)
  \]}
\end{frame}

\begin{frame}{Construindo a norma do produto vetorial}
Considere
\[
\vec v=(v_1,v_2,v_3)
\]
\[
\vec w=(w_1,w_2,w_3)
\]
Pela definição
\[
\vec v\times\vec w
=
\left(
v_2w_3-v_3w_2
,
v_3w_1-v_1w_3
,
v_1w_2-v_2w_1
\right)
\]
Queremos determinar
\[
\|\vec v\times\vec w\|
\]
\end{frame}

%------------------------------------------------------------------------------%
\begin{frame}{Calculando o quadrado da norma}
É conveniente calcular primeiro o quadrado da norma
\[
\|\vec v\times\vec w\|^2
\]
Temos
\[
\begin{aligned}
\|\vec v\times\vec w\|^2
={}&
(v_2w_3-v_3w_2)^2\\
&+
(v_3w_1-v_1w_3)^2\\
&+
(v_1w_2-v_2w_1)^2
\end{aligned}
\]
Vamos desenvolver os três quadrados
\end{frame}

%------------------------------------------------------------------------------%
\begin{frame}{Desenvolvendo os quadrados}
\[
\begin{aligned}
\|\vec v\times\vec w\|^2
={}&
v_2^2w_3^2
+
v_3^2w_2^2
-
2v_2v_3w_2w_3\\
&+
v_3^2w_1^2
+
v_1^2w_3^2
-
2v_3v_1w_3w_1\\
&+
v_1^2w_2^2
+
v_2^2w_1^2
-
2v_1v_2w_1w_2
\end{aligned}
\]
Os termos podem ser reorganizados para revelar uma identidade importante
\end{frame}

%------------------------------------------------------------------------------%
\begin{frame}{Uma identidade importante}
Observe que
\[
\|\vec v\|^2
=
v_1^2+v_2^2+v_3^2
\]
\[
\|\vec w\|^2
=
w_1^2+w_2^2+w_3^2
\]
Portanto
\[
\begin{aligned}
\|\vec v\|^2\|\vec w\|^2
={}&
v_1^2w_1^2+
v_1^2w_2^2+
v_1^2w_3^2\\
&+
v_2^2w_1^2+
v_2^2w_2^2+
v_2^2w_3^2\\
&+
v_3^2w_1^2+
v_3^2w_2^2+
v_3^2w_3^2
\end{aligned}
\]
\end{frame}

%------------------------------------------------------------------------------%
\begin{frame}{O papel do produto escalar}
Temos
\[
\vec v\cdot\vec w
=
v_1w_1+v_2w_2+v_3w_3
\]
Logo
\[
(\vec v\cdot\vec w)^2
=
v_1^2w_1^2+
v_2^2w_2^2+
v_3^2w_3^2
\]
\[
+
2v_1v_2w_1w_2
+
2v_1v_3w_1w_3
+
2v_2v_3w_2w_3
\]
Comparando as duas expressões obtemos
\[
\boxed{
\|\vec v\times\vec w\|^2
=
\|\vec v\|^2\|\vec w\|^2
-
(\vec v\cdot\vec w)^2
}
\]
\end{frame}

%------------------------------------------------------------------------------%
\begin{frame}{Extraindo a raiz}
Como uma norma é sempre não negativa
\[
\|\vec v\times\vec w\|\geq 0
\]
podemos escrever
\[
\|\vec v\times\vec w\|
=
\sqrt{
\|\vec v\|^2\|\vec w\|^2
-
(\vec v\cdot\vec w)^2
}
\]
Esta é uma fórmula para calcular diretamente a norma do produto vetorial
\end{frame}

%------------------------------------------------------------------------------%
\begin{frame}{Introduzindo o ângulo}
Do produto escalar sabemos que
\[
\vec v\cdot\vec w
=
\|\vec v\|\|\vec w\|\cos\theta
\]
Substituindo na expressão anterior
\[
\begin{aligned}
\|\vec v\times\vec w\|
&=
\sqrt{
\|\vec v\|^2\|\vec w\|^2
-
\|\vec v\|^2\|\vec w\|^2\cos^2\theta
}
\end{aligned}
\]
Colocando o fator comum em evidência
\[
\|\vec v\times\vec w\|
=
\sqrt{
\|\vec v\|^2\|\vec w\|^2
(1-\cos^2\theta)
}
\]
\end{frame}

%------------------------------------------------------------------------------%
\begin{frame}{Aparece o seno}
Pela identidade trigonométrica
\[
1-\cos^2\theta=\sin^2\theta
\]
temos
\[
\|\vec v\times\vec w\|
=
\sqrt{
\|\vec v\|^2
\|\vec w\|^2
\sin^2\theta
}
\]
Como
\[
\|\vec v\|\geq0
\]
\[
\|\vec w\|\geq0
\]
\[
0\leq\theta\leq\pi
\]
temos
\[
\sin\theta\geq0
\]
Portanto
\[
\boxed{
\|\vec v\times\vec w\|
=
\|\vec v\|\|\vec w\|\sin\theta
}
\]
\end{frame}

%------------------------------------------------------------------------------%
\begin{frame}{Duas formas equivalentes}
A norma do produto vetorial pode ser calculada de duas maneiras
\[
\|\vec v\times\vec w\|
=
\sqrt{
\|\vec v\|^2\|\vec w\|^2
-
(\vec v\cdot\vec w)^2
}
\]
ou
\[
\|\vec v\times\vec w\|
=
\|\vec v\|\|\vec w\|\sin\theta
\]
A primeira é particularmente útil quando conhecemos as coordenadas dos vetores

A segunda evidencia a interpretação geométrica
\end{frame}

%------------------------------------------------------------------------------%
\begin{frame}{Interpretação geométrica}
A expressão
\[
\|\vec v\|\|\vec w\|\sin\theta
\]
pode ser interpretada como
\[
\text{base}\times\text{altura}
\]
Escolhendo $\vec v$ como base
\[
\text{base}=\|\vec v\|
\]
\[
\text{altura}=\|\vec w\|\sin\theta
\]
Logo
\[
\|\vec v\times\vec w\|
=
\text{área do paralelogramo}
\]

\end{asy}
\end{center}
\end{frame}

%------------------------------------------------------------------------------%
\begin{frame}{Um caso particular}
Se
\[
\vec v\perp\vec w
\]
então
\[
\theta=\frac{\pi}{2}
\]
\[
\sin\theta=1
\]
Portanto
\[
\|\vec v\times\vec w\|
=
\|\vec v\|\|\vec w\|
\]
Neste caso o paralelogramo é um retângulo
\end{frame}


\begin{frame}{Outro caso particular}
Se os vetores são paralelos
\[
\theta=0
\]
ou
\[
\theta=\pi
\]
então
\[
\sin\theta=0
\]
Logo
\[
\vec v\times\vec w=\vec 0
\]
Geometricamente
\[
\text{área}=0
\]
Os dois vetores não determinam um paralelogramo não degenerado
\end{frame}
