%------------------------------------------------------------------------------%
\begin{question}
  Dados
  \[
    \vec{u} = (1, 2)
  \]
  \[
    \vec{v} = (3, -1)
  \]
  Determine
  \[
    \vec{w} = 2\vec{u} - 3\vec{v}
  \]
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
  \[
    2\vec{u}
    \pause
    \;=\; 2(1, 2)
    \pause
    \;=\; (2\times 1, \; 2\times 2)
    \pause
    \;=\; (2, 4)
  \]

  \pause
  \[
    3\vec{v}
    \pause
    \;=\; 3(3, -1)
    \pause
    \;=\; (3\time 3, \; 3(-1))
    \pause
    \;=\; (9, -3)
  \]

  \pause
  Portanto
  \[
    \pause
    \vec{w}
    \pause
    \;=\; 2\vec{u}-3\vec{v}
    \pause
    \;=\; (2, 4) - (9, -3)
    \pause
    \;=\; (-7, 7)
  \]
\end{resolution}

\endinput
