\begin{question}
  Determine todos os vetores simultaneamente perpendiculares
  a $\vec{v}$ e $\vec{w}$ \\
  e que possuem módulo 10
  \[
    \vec{v} = \vetortri{1}{2}{3}
    \qquad
    \vec{w} = \vetortri{2}{-1}{1}
  \]
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Produto Vetorial}
\begin{align*}
  \onslide<+->{\vec{\omega}}
  \onslide<+->{& = \vec{v}\times\vec{w}}
  \onslide<+->{\;=\; \begin{vmatrix}
    \vec{i} & \vec{j} & \vec{k} \\
     1      &   2     &  3      \\
     2      &  -1     &  1
  \end{vmatrix}}
  \\
  \onslide<+->{& = \begin{vmatrix}
       2 &  3 \\
      -1 &  1
    \end{vmatrix}
    \vec{i}
    \;-\;
    \begin{vmatrix}
       1 &  3 \\
       2 &  1
    \end{vmatrix}
    \vec{j}
    \;+\;
    \begin{vmatrix}
       1 &  2 \\
       2 & -1
    \end{vmatrix}
    \vec{k}}
  \\
  \onslide<+->{
  & = \big( 2 \times 1 - 3(-1) \big) \vec{i}
    - \big( 1 \times 1 - 3 \times 2 \big) \vec{j}
    + \big( 1(-1) - 2 \times 2 \big) \vec{k}}
  \\
  \onslide<+->{& = 5 \vec{i} -5 \vec{j} - 3 \vec{k}}
  \onslide<+->{\;=\; \vetortri{5}{-5}{-3}}
\end{align*}
\end{resolution}

%------------------------------------------------------------------------------%
\begin{resolution}{Vetor Unitário}
\begin{columns}[t]
\column{0.3\textwidth}
  \onslide<+->{Norma de \(\vec{\omega}\)}
  \begin{align*}
    \onslide<+->{\norm{\vec{\omega}}}
    \onslide<+->{& = \norm{\vetortri{5}{-5}{-3}} \\}
    \onslide<+->{& = \sqrt{5^2 + (-5)^2 + (-3)^2} \\}
    \onslide<+->{& = \sqrt{25 + 25 + 9} \\}
    \onslide<+->{& = \sqrt{59}}
  \end{align*}
\column{0.5\textwidth}
  \onslide<+->{Vetor unitário da direção e sentido de \(\vec{\omega}\)}
  \begin{align*}
    \onslide<+->{\hat{\omega}}
    \onslide<+->{& = \frac{\vec{\omega}}{\norm{\vec{\omega}}} }
    \onslide<+->{\;=\; \frac{1}{\sqrt{59}} \vetortri{5}{-5}{-3} \\}
    \onslide<+->{& = \Vetortri{\frac{5}{\sqrt{59}}}{\frac{-5}{\sqrt{59}}}{\frac{-3}{\sqrt{59}}}}
  \end{align*}
\end{columns}
\end{resolution}
