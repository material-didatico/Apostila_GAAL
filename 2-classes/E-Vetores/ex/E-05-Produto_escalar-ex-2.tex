\begin{question}
  Calcule o produto escalar dos vetores
  \[
    \vec{u} = \vetortri{1}{-2}{3}
    \qquad\qquad
    \vec{v} = \vetortri{4}{0}{-1}
  \]

\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
  \begin{align*}
    \onslide<+->{\vec{u} \cdot \vec{v}}
    \onslide<+->{& = \vetortri{1}{-2}{3} \cdot \vetortri{4}{0}{-1} \\}
    \onslide<+->{& = 1 \times 4 + (-2) 0 + 3(-1) \\}
    \onslide<+->{& = 4 + 0 - 3 \\}
    \onslide<+->{& = 1}
  \end{align*}
\end{resolution}
