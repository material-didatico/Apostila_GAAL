\begin{question}
  Dado
  \[
    \vec{u} = ( 2, -3 )
  \]
  Determine
  \[
    3\vec{u}
    \qquad\text{e}\qquad
    -2\vec{u}
  \]
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
\begin{columns}
\column{0.3\linewidth}
  \begin{align*}
    \onslide<+->{\vec{v}}
    \onslide<+->{& = 3\vec{u}      \\}
    \onslide<+->{& = 3 ( 2, -3 )   \\}
    \onslide<+->{& = ( 6, -9 )          }
  \end{align*}
\column{0.5\linewidth}
  \begin{align*}
    \onslide<+->{\vec{w}}
    \onslide<+->{& = -2\vec{u}     \\}
    \onslide<+->{& = -2 ( 2, -3 )  \\}
    \onslide<+->{& = ( -4, 6 )}
  \end{align*}
\end{columns}
\end{resolution}
