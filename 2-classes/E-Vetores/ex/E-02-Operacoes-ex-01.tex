%------------------------------------------------------------------------------%
\begin{question}
  Determine o módulo do vetor
  \(
    \vec{u} = ( 6, 8 )
  \)
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
  \begin{align*}
  \onslide<+->{\norm{\vec{u}}}
  \onslide<+->{& = \sqrt{ 6^2 + 8^2 } \\}
  \onslide<+->{& = \sqrt{ 36 + 64 } \\}
  \onslide<+->{& = \sqrt{ 100 } \\}
  \onslide<+->{& = 10}
  \end{align*}
\end{resolution}
