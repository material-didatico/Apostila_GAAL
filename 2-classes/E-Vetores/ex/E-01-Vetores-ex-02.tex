\begin{question}
  Encontre o vetor $\vec{v} = \overrightarrow{AB}$ determinado pelos pontos
  \[
    A = ( 2, 3 )
  \]
  \[
    B = ( 1, 4 )
  \]
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
  O deslocamento na coordenada $x$ entre os pontos $A$ e $B$ é
  \pause
  \[
    v_x
    \pause
    \;=\; 1 - 2
    \pause
    \;=\; -1
  \]
  \pause
  O deslocamento na coordenada $y$ entre os pontos $A$ e $B$ é
  \[
    \pause
    v_y
    \pause
    \;=\; 4 - 3
    \pause
    \;=\; 1
  \]
  \pause
  Portanto
  \[
    \pause
    \vec{v}
    \pause
    \;=\; \vetor{v_x}{v_y}
    \pause
    \;=\; \vetor{-1}{1}
  \]
\end{resolution}

%------------------------------------------------------------------------------%
\endinput
