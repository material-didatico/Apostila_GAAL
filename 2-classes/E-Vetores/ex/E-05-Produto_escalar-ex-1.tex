\begin{question}
  Calcule o produto escalar dos vetores
  \[
    \vec{u} = ( 2, 3 )
    \qquad\qquad
    \vec{v} = ( 4, -1 )
  \]
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
  \begin{align*}
    \onslide<+->{\vec{u} \cdot \vec{v}}
    \onslide<+->{& = ( 2, 3 ) \cdot ( 4, -1 ) \\}
    \onslide<+->{& = 2 \times 4 + 3(-1) \\}
    \onslide<+->{& = 8 - 3 \\}
    \onslide<+->{& = 5}
  \end{align*}
\end{resolution}
