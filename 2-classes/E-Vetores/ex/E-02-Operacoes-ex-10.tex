\begin{question}
  Avalie
  \(
    \norm{\vec{u} + 2 \vec{v}}
  \)
  para
  \[
    \vec{u} = \vetortri{2}{-1}{3}
  \]
  \[
    \vec{v} = \vetortri{1}{4}{-2}
  \]
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
  \[
    2\vec{v}
    \pause
    \;=\; 2 \vetortri{2}{-1}{3}
    \pause
    \;=\; \vetortri{2\times 2}{2(-1)}{2\times 3}
    \pause
    \;=\; \vetortri{2}{8}{-4}
  \]
  \pause
  \[
    \vec{u} + 2 \vec{v}
    \pause
    \;=\; \vetortri{2}{-1}{3} + \vetortri{2}{8}{-4}
    \pause
    \;=\; \vetortri{4}{7}{-1}
  \]
  \pause
  \[
    \norm{\vec{u} + 2 \vec{v}}
    \pause
    \;=\; \norm{\vetortri{4}{7}{-1}}
    \pause
    \;=\; \sqrt{4^2 + 7^2 + (-1)^2\,}
    \pause
    \;=\; \sqrt{16 + 49 + 1\,}
    \pause
    \;=\; \sqrt{66}
  \]
\end{resolution}
