\begin{question}
  Dados
  \[
    \vec{u} = (3, 4)
  \]
  \[
    \vec{v} = (4, -3)
  \]
  Determine o módulo da soma dos vetores
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
\begin{columns}[t]
\column{0.5\linewidth}
  \onslide<+->{Calculando a Soma}
  \begin{align*}
    \onslide<+->{\vec{u} + \vec{v}}
    \onslide<+->{& = (3, 4) + (4, -3) \\}
    \onslide<+->{& = (3+4,\; 4-3) \\}
    \onslide<+->{& = (7, 1)}
  \end{align*}
\column{0.5\linewidth}
  \onslide<+->{Calculando o módulo}
  \begin{align*}
    \onslide<+->{\norm{\vec{u}+\vec{v}}}
    \onslide<+->{& = \norm{(7, 1)} \\}
    \onslide<+->{& = \sqrt{7^2+1^2}\\}
    \onslide<+->{& = \sqrt{50}\\}
    \onslide<+->{& = 5\sqrt{2}}
  \end{align*}
\end{columns}
\end{resolution}
