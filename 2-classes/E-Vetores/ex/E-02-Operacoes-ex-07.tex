\begin{question}
  Dados
  \[
    \vec{u} = ( 3, -2 )
  \]
  \[
    \vec{v} = ( -1, 5 )
  \]
  Determine
  \[
    \vec{w} = \vec{u} + \vec{v}
  \]

\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
  \begin{align*}
    \onslide<+->{\vec{w}}
    \onslide<+->{& = \vec{u} + \vec{v} \\}
    \onslide<+->{& = \vetor{3}{-2} + \vetor{-1}{5}  \\}
    \onslide<+->{& = \vetor{3+(-1)}{-2+5} \\}
    \onslide<+->{& = \vetor{2}{3}}
  \end{align*}
\end{resolution}
