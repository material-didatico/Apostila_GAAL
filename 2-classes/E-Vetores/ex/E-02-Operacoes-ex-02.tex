\begin{question}
  Determine o valor de $k$ para que o vetor
  \[
    \vec{u} = \vetor{k}{4}
  \]
  possua módulo igual a 5
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
  \begin{align*}
    \onslide<+->{\norm{\vec{u}} = \sqrt{ k^2 + 16 } & = 5                          \\}
    \onslide<+->{k^2 + 16                           & = 25                         \\}
    \onslide<+->{k^2                                & = 25 - 16                    \\}
    \onslide<+->{k^2                                & = 9                          \\}
    \onslide<+->{\abs{k}                            & =3                           \\}
    \onslide<+->{k                                  & =\pm3                        \\}
    \onslide<+->{k=3                                & \quad \text{ou}\quad k=-3}
  \end{align*}
\end{resolution}
