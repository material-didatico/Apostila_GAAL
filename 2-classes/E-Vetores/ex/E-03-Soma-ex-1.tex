\begin{question}
  Calcule a soma dos vetores
  \[
    \vec{w} = (2, 3)
  \]
  \[
    \vec{v} = (4, -1)
  \]

\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
  \begin{align*}
    \onslide<+->{\vec{w}+\vec{v}}
    \onslide<+->{& = (2, 3) + (4, -1) \\}
    \onslide<+->{& = (2+4,\, 3-1) \\}
    \onslide<+->{& = (6, 2)}
  \end{align*}
\end{resolution}
