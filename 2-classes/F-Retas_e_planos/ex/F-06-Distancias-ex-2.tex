\begin{question}
  Calcule a distância entre
  \[
    A = (-2, 3)
  \]
  \[
    B = (4, -5)
  \]
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
\begin{columns}[t]
\column{0.5\textwidth}
  \begin{align*}
    \onslide<+->{\overrightarrow{AB} & = B - A             \\}
    \onslide<+->{                    & = (-2, 3) - (4, -5) \\}
    \onslide<+->{                    & = (6, -8)}
  \end{align*}
\column{0.5\textwidth}
  \begin{align*}
    \onslide<+->{d(A, B) & = \norm{\overrightarrow{AB}} \\}
    \onslide<+->{        & = \norm{(6, -8)}             \\}
    \onslide<+->{        & = \sqrt{6^{2}+(-8)^{2}}      \\}
    \onslide<+->{        & = \sqrt{36+64}               \\}
    \onslide<+->{        & = \sqrt{100}                 \\}
    \onslide<+->{        & = 10}
  \end{align*}
\end{columns}
\end{resolution}
