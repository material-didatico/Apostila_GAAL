\begin{question}
  Determine a equação simétrica da reta
  \[
    r\colon
    \begin{cases}
      x =  2 + 3t \\
      y = -1 - 2t \\
      z =  4 +  t
    \end{cases}
  \]
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
  Isolando o parâmetro em cada equação
  \[
    \pause
    t =  \frac{x-2}{3} \qquad\pause
    t = -\frac{y+1}{2} \qquad\pause
    t = z-4
  \]
  \pause
  temos
  \[
    \frac{x-2}{3}
    = \frac{y+1}{-2}
    = z-4
  \]
\end{resolution}
