\begin{question}
  Calcule a distância entre
  \[
    A = (1, -2, 3)
  \]
  \[
    B = (4, 2, -1)
  \]
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
\begin{columns}[t]
\column{0.5\textwidth}
  \begin{align*}
    \onslide<+->{\overrightarrow{AB} & = B - A                   \\}
    \onslide<+->{                    & = (4, 2, -1) - (1, -2, 3) \\}
    \onslide<+->{                    & = (4-1, 2-(-2), -1-3)     \\}
    \onslide<+->{                    & = (3, 4, -4)}
  \end{align*}
\column{0.5\textwidth}
  \begin{align*}
    \onslide<+->{d(A, B) & = \norm{\overrightarrow{AB}}  \\}
    \onslide<+->{        & = \norm{(3, 4, -4)}           \\}
    \onslide<+->{        & = \sqrt{3^{2}+4^{2}+(-4)^{2}} \\}
    \onslide<+->{        & = \sqrt{9+16+16}              \\}
    \onslide<+->{        & = \sqrt{41}}
  \end{align*}
\end{columns}
\end{resolution}
