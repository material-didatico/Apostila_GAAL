\begin{question}
  Determine uma equação vetorial e as equações paramétricas do plano que passa pelo ponto
  \[
    A = (1, 2, -1)
  \]
  e contém os vetores
  \[
    \vec{v} = (2, -1, 3)
  \]
  \[
    \vec{w} = (1, 2, 1)
  \]
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
  A equação vetorial do plano é
  \[
    \vec{r} = A + s\vec{v} + t\vec{w}
  \]
  \pause
  \[
    \vetortri{x}{y}{z}
    = \vetortri{1}{2}{-1}
    + s\vetortri{2}{-1}{3}
    + t\vetortri{1}{2}{1}
    \qquad\qquad
    s, \, t \in \R
  \]
  \pause
  Separando as coordenadas
  \[
    \begin{cases}
      x =  1 + 2s +  t  \\
      y =  2 -  s + 2t  \\
      z = -1 + 3s + t
    \end{cases}
  \]
\end{resolution}
