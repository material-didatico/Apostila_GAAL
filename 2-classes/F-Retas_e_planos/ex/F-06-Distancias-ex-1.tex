\begin{question}
  Calcule a distância entre
  \[
    A = (1, 2)
  \]
  \[
    B = (5, 5)
  \]

\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
\begin{columns}[t]
\column{0.5\textwidth}
  \begin{align*}
    \onslide<+->{\overrightarrow{AB} & = B - A           \\}
    \onslide<+->{                    & = (5, 5) - (1, 2) \\}
    \onslide<+->{                    & = (5-1, 5-2)      \\}
    \onslide<+->{                    & = (4, 3)}
  \end{align*}
\column{0.5\textwidth}
  \begin{align*}
    \onslide<+->{d(A, B) & = \norm{\overrightarrow{AB}} \\}
    \onslide<+->{        & = \norm{(4, 3)}              \\}
    \onslide<+->{        & = \sqrt{4^{2}+3^{2}}         \\}
    \onslide<+->{        & = \sqrt{16+9}                \\}
    \onslide<+->{        & = \sqrt{25}                  \\}
    \onslide<+->{        & = 5}
  \end{align*}
\end{columns}
\end{resolution}
