%------------------------------------------------------------------------------%
\begin{question}
  Determine uma parametrização para o plano
  \[
    2x - y + z - 3 = 0
  \]
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
\begin{columns}[t]
\column{0.5\textwidth}
  Temos
  \[
    2x-y+z-3=0
  \]
  \pause
  Isolamos uma coordenada
  \[
    y=2x+z-3
  \]
  \pause
  Escolhemos duas variáveis livres
  \[
    x = s
  \]
  \[
    z = t
  \]
\column{0.5\textwidth}
  \pause
  Equação parametrica do plano
  \[
    \begin{cases}
      x = s \\
      y = -3 +2s + t \\
      z = t
    \end{cases}
  \]
  com $s$, $t\in\mathbb{R}$
\end{columns}
\end{resolution}
