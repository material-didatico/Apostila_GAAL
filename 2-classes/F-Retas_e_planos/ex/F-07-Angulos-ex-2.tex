\begin{question}
  Determine o ângulo entre as retas
  \[
    r\colon
    \begin{cases}
      x=1+t  \\
      y=2+2t \\
      z=3-t
    \end{cases}
  \qquad\qquad
    s\colon
    \begin{cases}
      x = 2 + \tau  \\
      y = 1 - \tau  \\
      z = 4 +2\tau
    \end{cases}
  \]
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
  Vetores diretores
  \begin{align*}
    \vec{v} & = (1,  2, -1) \\
    \vec{w} & = (1, -1, 2)
  \end{align*}

  Produto escalar
  \begin{align*}
    \vec{v}\cdot\vec{w} & = 1\cdot1+2\cdot(-1)+(-1)\cdot2 \\
                        & = 1-2-2                         \\
                        & =-3
  \end{align*}
\end{resolution}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
  Normas
  \begin{align*}
    \norm{\vec{v}} & = \sqrt{1^2+2^2+(-1)^2}
                     = \sqrt{6}
  \end{align*}

  \begin{align*}
    \norm{\vec{w}} & = \sqrt{1^2+(-1)^2+2^2}
                     = \sqrt{6}
  \end{align*}

  \begin{align*}
    \cos\theta & = \frac{|-3|} {\sqrt{6}\sqrt{6}}
                 = \frac{3}{6}
                 = \frac{1}{2}
  \end{align*}

  \[
    \theta=\frac{\pi}{3}
  \]
\end{resolution}
