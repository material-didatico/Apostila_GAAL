\begin{question}
  Determine o ângulo entre as retas
  \[
    r\colon
    \begin{cases}
      x=1+t  \\
      y=2+2t \\
      z=3-t
    \end{cases}
  \qquad\qquad
    s\colon
    \begin{cases}
      x = 2 + \tau  \\
      y = 1 - \tau  \\
      z = 4 +2\tau
    \end{cases}
  \]
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
  \onslide<+->{Vetores diretores}
  \begin{align*}
    \onslide<+->{\vec{v} & = (1,  2, -1) \\}
    \onslide<+->{\vec{w} & = (1, -1, 2)}
  \end{align*}

  \onslide<+->{Produto escalar}
  \begin{align*}
    \onslide<+->{\vec{v}\cdot\vec{w}}
    \onslide<+->{& = 1\cdot1+2\cdot(-1)+(-1)\cdot2 \\}
    \onslide<+->{& = 1-2-2                         \\}
    \onslide<+->{& =-3}
  \end{align*}
\end{resolution}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
  \onslide<+->{Normas}
  \begin{align*}
    \onslide<+->{\norm{\vec{v}}}
    \onslide<+->{& = \sqrt{1^2+2^2+(-1)^2}}
    \onslide<+->{  = \sqrt{6}}
  \end{align*}

  \begin{align*}
    \onslide<+->{\norm{\vec{w}}}
    \onslide<+->{& = \sqrt{1^2+(-1)^2+2^2}}
    \onslide<+->{  = \sqrt{6}}
  \end{align*}

  \begin{align*}
    \onslide<+->{\cos\theta}
    \onslide<+->{& = \frac{|-3|}{\sqrt{6}\sqrt{6}}}
    \onslide<+->{  = \frac{3}{6}}
    \onslide<+->{  = \frac{1}{2}}
  \end{align*}

  \[
    \onslide<+->{\theta=\frac{\pi}{3}}
  \]
\end{resolution}
