\begin{question}
  Determine o ângulo entre os planos
  \[
    \alpha\colon x + 2y + 2z - 1 = 0
  \]
  \[
    \beta\colon 2x - y + 2z + 3 = 0
  \]
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
  Vetores normais
  \[
    \vec{n}_\alpha = (1, 2, 2)
  \]
  \[
    \vec{n}_\beta = (2, -1, 2)
  \]

  Produto escalar
  \begin{align*}
    \vec{n}_\alpha\cdot\vec{n}_\beta & = 1\cdot2+2\cdot(-1)+2\cdot2 \\
                                     & = 2-2+4                      \\
                                     & = 4
  \end{align*}
\end{resolution}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
  Normas
  \begin{align*}
    \norm{\vec{n}_1} & = \sqrt{1^2+2^2+2^2} \\
                     & = 3
  \end{align*}

  \begin{align*}
    \norm{\vec{n}_2} & = \sqrt{2^2+(-1)^2+2^2} \\
                     & = 3
  \end{align*}
\end{resolution}
