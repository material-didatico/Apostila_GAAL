\begin{question}
  Determine o ângulo entre os planos
  \[
    \alpha\colon x + 2y + 2z - 1 = 0
  \]
  \[
    \beta\colon 2x - y + 2z + 3 = 0
  \]
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
  \onslide<+->{Vetores normais}
  \[
    \onslide<+->{\vec{n}_\alpha = (1, 2, 2)}
  \]
  \[
    \onslide<+->{\vec{n}_\beta = (2, -1, 2)}
  \]

  \onslide<+->{Produto escalar}
  \begin{align*}
    \onslide<+->{\vec{n}_\alpha\cdot\vec{n}_\beta}
    \onslide<+->{& = 1\cdot2+2\cdot(-1)+2\cdot2 \\}
    \onslide<+->{& = 2-2+4                      \\}
    \onslide<+->{& = 4}
  \end{align*}
\end{resolution}
