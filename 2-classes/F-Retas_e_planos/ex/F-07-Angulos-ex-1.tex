\begin{question}
Determine o ângulo entre os planos
  \[
    \alpha\colon x + y + z - 1 = 0
  \]
  \[
    \beta\colon 2x - y + 2z + 3 = 0
  \]
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
  Vetores normais
  \[
    \vec{n}_{1} = (1,  1, 1)
    \qquad\qquad
    \vec{n}_{2} = (2, -1, 2)
  \]

  \[
    \pause
    \vec{n}_{1}\cdot\vec{n}_{2}
    \pause
    = 2 - 1 + 2
    \pause
    = 3
  \]

  \[
    \pause
    \norm{\vec{n}_{1}}
    \pause
    = \sqrt{3}
  \]

  \[
    \pause
    \norm{\vec{n}_{2}}
    \pause
    = \sqrt{4 + 1 + 4}
    \pause
    = 3
  \]

  \[
    \pause
    \cos{\theta}
    \pause
    = \frac{3}{3\sqrt{3}}
    \pause
    = \frac{1}{\sqrt{3}}
  \]

  \[
    \pause
    \theta
    = \arccos{\left(\frac{1}{\sqrt{3}}\right)}
  \]
\end{resolution}
