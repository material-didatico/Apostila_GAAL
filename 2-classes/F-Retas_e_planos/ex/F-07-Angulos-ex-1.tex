\begin{question}
  Determine o ângulo entre
  \[
    \vec{v} = (1, 2, 2)
  \]
  \[
    \vec{w} = (2, -1, 2)
  \]
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
  \begin{align*}
    \vec{v}\cdot\vec{w} & = 1\cdot2+2\cdot(-1)+2\cdot2 \\
                        & = 2-2+4 = 4
  \end{align*}

  \begin{align*}
    \norm{\vec{v}} & = \sqrt{1^2+2^2+2^2} = 3 \\
    \norm{\vec{w}} & = \sqrt{2^2+(-1)^2+2^2} = 3
  \end{align*}

  \begin{align*}
    \cos\theta & = \frac{4}{3\cdot3} = \frac{4}{9}
  \end{align*}

  \[
    \theta=\arccos\left(\frac{4}{9}\right)
  \]

\end{resolution}
