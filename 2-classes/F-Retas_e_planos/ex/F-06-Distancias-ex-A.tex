\begin{question}
  Calcule a distância do ponto
  \[
    P = (2, -1, 3)
  \]
  ao plano
  \[
  \gamma\colon 2x - y + 2z - 5 = 0
  \]
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
  \onslide<+->{Vetor normal do plano}
  \[
    \onslide<+->{\vec{n} = (2, -1, 2)}
  \]
  \[
    \onslide<+->{\norm{\vec{n}}}
    \onslide<+->{= \sqrt{2^{2}+(-1)^{2}+2^{2}}}
    \onslide<+->{= 3}
  \]

  \onslide<+->{Expressão do plano no ponto $P = (2, -1, 3)$}
  \begin{align*}
    \onslide<+->{2x_0 - y_0 + 2z_0 - 5}
    \onslide<+->{& = 2(2) - (-1) + 2(3) - 5 \\}
    \onslide<+->{& = 4 + 1 + 6 - 5          \\}
    \onslide<+->{& = 6}
  \end{align*}
\end{resolution}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
  Distância
  \[
    d(P, \gamma)
    \pause
    = \frac{\, \abs{ax_0 + by_0 + cz_0 + d} \,}{\norm{\vec{n}}}
    \pause
    = \frac{\abs{6}}{3}
    \pause
    = 2
  \]
\end{resolution}
