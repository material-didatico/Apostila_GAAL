B=(6,-2,5)

\begin{frame}{Exemplo 6 --- Solução}
\begin{align*}
M
&=
\left(
\frac{-2+6}{2},
\frac{4+(-2)}{2},
\frac{1+5}{2}
\right)
\\
&=
\left(
\frac{4}{2},
\frac{2}{2},
\frac{6}{2}
\right)
\\
&=
(2,1,3)
\end{align*}
