%------------------------------------------------------------------------------%
\begin{question}
  Determine se o ponto
  \(
    P = (0, 7, 8)
  \)
  pertence à reta
  \[
    r\colon
    \begin{cases}
      x = 2 -  t \\
      y = 1 + 3t \\
      z = 4 + 2t
    \end{cases}
  \]
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
  \onslide<+->{Para que $P$ pertença à reta,}
  \onslide<+->{deve existir um mesmo valor de $t\in\R$} \\
  \onslide<+->{que satisfaça as três equações}
\begin{columns}
\column{0.3\textheight}
  \begin{align*}
    \onslide<+->{x & = 2 - t \\}
    \onslide<+->{0 & = 2 - t \\}
    \onslide<+->{t & = 2}
  \end{align*}
\column{0.3\textheight}
  \begin{align*}
    \onslide<+->{y & = 1 + 3t \\}
    \onslide<+->{  & = 1 + 3\times 2 \\}
    \onslide<+->{  & = 7}
  \end{align*}
\column{0.3\textheight}
  \begin{align*}
    \onslide<+->{z & = 4 + 2t \\}
    \onslide<+->{  & = 4 + 2\times 2 \\}
    \onslide<+->{  & = 8}
  \end{align*}
\end{columns}

  \onslide<+->{As três coordenadas são satisfeitas pelo mesmo parâmetro,} \\
  \onslide<+->{portanto, $P\in r$}
\end{resolution}
