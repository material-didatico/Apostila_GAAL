\begin{question}
  Escreva a equação
  \[
    3x^2 + 2xy + 2y^2 - 4x + 6y = 1
  \]
  na forma matricial
  \(
    v^tAv + B^tv + f = 0
  \)

  \bigskip
  Determine $A$, $B$ e $f$
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
\begin{columns}[t]
\column{0.5\textwidth}
  Equação
  \[
    3x^2 + 2xy + 2y^2 - 4x + 6y = 1
  \]

  \pause
  Parte quadrática
  \[
    \pause
    ax^2 + 2bxy + cy^2
    \pause
    = 3x^2 + 2xy + 2y^2
  \]
  \[
    \pause
    A = \begin{bmatrix}
      3 & 1 \\
      1 & 2
    \end{bmatrix}
  \]

\column{0.4\textwidth}
  \pause
  Termos lineares
  \[
    \pause
    -4x + 6y
    \pause
    =
    \begin{bmatrix} -4 & 6  \end{bmatrix}
    \begin{bmatrix}  x \\ y \end{bmatrix}
  \]
  \[
    \pause
    B = \begin{bmatrix} -4 \\ 6 \end{bmatrix}
  \]

  \bigskip
  \pause
  Termo constante
  \[
    \pause
    f = -1
  \]
\end{columns}
\end{resolution}
