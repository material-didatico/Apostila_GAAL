\begin{question}
  Descreva o conjunto de soluções reais da equação
  \[
  x^2+y^2+1=0
  \]
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
  Para todos os reais \(x\) e \(y\)
  \[
    x^2 + y^2 \geq 0
  \]
  \[
    \pause
    x^2 + y^2 + 1 \geq 1
  \]

  \pause
  A expressão não pode ser igual a zero

  \pause
  Portanto, a equação não possui soluções reais
\end{resolution}
