\begin{question}
  Considerando a equação
  \[
    x^2 + 2y = 0
  \]
  \vspace{-\baselineskip}
  \begin{enumerate}
    \item identifique a curva
    \item determine seus principais valores e pontos
    \item esboce a curva
  \end{enumerate}
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Identificação}
  Comparando
  \tabto{30mm}
  \(
    x^2 = -2y
  \)

  \pause
  com
  \tabto{30mm}
  \(
    x^2 = 4py
  \)

  \pause
  A curva é uma parábola
  \pause
  com vértice em $(0, 0)$
  \[
    \pause
    4p = -2
    \qquad
    \qquad
    \pause
    p = \frac{-2}{4}
    \pause
    = -\frac{1}{2}
  \]
  \pause
  O termo elevado ao quadrado é $x^2$

  \pause
  O eixo de simetria é paralelo ao eixo $y$

  \pause
  Como $p < 0$ a parábola tem abertura para a baixo
\end{resolution}
