%------------------------------------------------------------------------------%
\begin{question}
  Descreva o conjunto de soluções reais da equação
  \[
    x^2 + y^2 = 0
  \]
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
\begin{columns}
\column{0.4\textwidth}
  Como
  \[
    x^2\geq 0
    \qquad
    y^2\geq 0
  \]
  \pause
  a igualdade
  \[
    x^2 + y^2 = 0
  \]
  \pause
  só pode ocorrer quando
  \begin{align*}
    \onslide<+->{x^2 & = 0 & y^2 & = 0} \\
    \onslide<+->{x   & = 0 & y   & = 0}
  \end{align*}
  \column{0.4\textwidth}
  \onslide<+->{O conjunto de soluções é
  \[
    s = \big\{ (0,0) \big\}
  \]}
\end{columns}
\end{resolution}
