\begin{question}
  Dada a equação da parábola
  \[
    (x - 3)^2 = 12(y + 1)
  \]
  \vspace{-1.5\baselineskip}
  \begin{enumerate}
    \item identifique vértice e o parâmetro $p$
    \item escreva a equação matricial
    \item esboce a curva
  \end{enumerate}
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Parâmetro $p$}
  Comparando
  \[
    (x - 3)^2 = 12(y + 1)
  \]
  \pause
  com a equação transladada
  \[
    (x-h)^2 = 4p(y-k)
  \]
  \pause
  identificamos
  \[
    \pause
    4p = 12
    \qquad
    \pause
    p = 3
  \]
  \pause
  vértice:
  \(
    \pause
    \mathcal{V}
    \pause
    = (h, k)
    \pause
    = (3, -1)
  \)
\end{resolution}

%------------------------------------------------------------------------------%
\begin{resolution}{Equação Matricial}
  Equação Matricial
  \[
    V^tAV + 2B^tV = 0
  \]
  \pause
  com
  \[
    \pause
    A = \begin{bmatrix}
      1 & 0 \\
      0 & 0
    \end{bmatrix}
    \qquad\qquad
    \pause
    B
    = \begin{bmatrix}[c]
      0   \\
      -2p
    \end{bmatrix}
    \pause
    = \begin{bmatrix}
      0  \\
      -6
    \end{bmatrix}
  \]
  \[
    \pause
    V
    \pause
    = \begin{bmatrix}
      x - h  \\
      y - k
    \end{bmatrix}
    \pause
    = \begin{bmatrix}[c]
      x - 3  \\
      y - (-1)
    \end{bmatrix}
    \pause
    = \begin{bmatrix}
      x - 3  \\
      y + 1
    \end{bmatrix}
  \]
\end{resolution}
