\begin{question}
  Dada a equação da elipse
  \[
    \frac{x^2}{9}+\frac{y^2}{4} = 1
  \]
  \vspace{-1.5\baselineskip}
  \begin{enumerate}
    \item identifique os semieixos ($a$ e $b$)
    \item escreva a equação matricial
    \item esboce a curva
  \end{enumerate}
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Semieixos $a$ e $b$}
  Comparando
  \[
    \frac{x^2}{9} + \frac{y^2}{4} = 1
  \]
  \pause
  com a equação canônica
  \[
    \frac{x^2}{a^2} + \frac{y^2}{b^2} = 1
  \]
  \pause
  identificamos
  \[
    \pause
    a^2 = 9
    \qquad
    \pause
    a = 3
  \]
  \[
    \pause
    b^2 = 4
    \qquad
    \pause
    b = 2
  \]
\end{resolution}

%------------------------------------------------------------------------------%
\begin{resolution}{Equação Matricial}
  Equação Matricial
  \[
    v^t A v = 1
  \]
  \pause
  com
  \[
    A
    = \begin{bmatrix}
      \dfrac{1}{a^2} & 0              \\[2mm]
      0              & \dfrac{1}{b^2}
    \end{bmatrix}
    \pause
    = \begin{bmatrix}
      \dfrac{1}{9} & 0            \\[2mm]
      0            & \dfrac{1}{4}
    \end{bmatrix}
  \]
\end{resolution}
