\begin{question}
  Considerando a equação
  \[
    9y^2 - 4x^2 = 36
  \]
  \vspace{-\baselineskip}
  \begin{enumerate}
    \item identifique a curva
    \item determine seus principais valores e pontos
    \item esboce a curva
  \end{enumerate}
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Identificação}
\begin{columns}
\column{0.45\textwidth}
  \begin{align*}
    \onslide<+->{9y^2            - 4x^2            & = 36 \\}
    \onslide<+->{\frac{9y^2}{36} - \frac{4x^2}{36} & = 1  \\}
    \onslide<+->{\frac{x^2}{4}   - \frac{x^2}{9}   & = 1}
  \end{align*}
  \onslide<+->{comparando com
   \[
     \frac{y^2}{a^2} - \frac{x^2}{b^2} = 1
   \]}
   \onslide<+->{Hipérbole com vértices no eixo $y$}
\column{0.4\textwidth}
  \onslide<+->{Centro: \(C = (0, 0)\)}

  \[
    \onslide<+->{a^2 = 4}
    \qquad
    \onslide<+->{a = 2}
  \]

  \[
    \onslide<+->{b^2 = 9}
    \qquad
    \onslide<+->{b = 3}
  \]
  \onslide<+->{Como o termo positivo é o $y^2$}

  \onslide<+->{os vértices estão no eixo $y$}

  \bigskip
  \onslide<+->{Vértices: $(0, -2)$ e $(0, 2)$}
\end{columns}
\end{resolution}
