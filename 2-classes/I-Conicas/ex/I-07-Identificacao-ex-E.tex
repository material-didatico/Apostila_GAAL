\begin{question}
  Considerando a equação
  \[
    x^2-4x-5=0
  \]
  \vspace{-\baselineskip}
  \begin{enumerate}
    \item identifique a curva
    \item determine seus principais valores e pontos
    \item esboce a curva
  \end{enumerate}
\end{question}

%------------------------------------------------------------------------------%
 \begin{resolution}{Completando quadrados}
   Completamos o quadrado na variável $x$

   \begin{align*}
     x^2-4x-5&=0
     \\
     x^2-4x+4&=5+4
     \\
     (x-2)^2&=9
   \end{align*}

   Extraindo a raiz quadrada

   \begin{align*}
     x-2&=\pm 3
     \\
     x&=2\pm 3
   \end{align*}

   Portanto

   \[
     x=5
     \qquad\text{ou}\qquad
     x=-1
   \]
 \end{resolution}

 %------------------------------------------------------------------------------%
 \begin{resolution}{Classificação}
   A equação representa duas retas

   \[
     x=5
     \qquad
     x=-1
   \]

   As duas retas são verticais e paralelas ao eixo $y$

   Portanto, trata-se de uma cônica degenerada formada
   por duas retas paralelas distintas
 \end{resolution}
