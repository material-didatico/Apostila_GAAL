\begin{question}
  Determine e esboce o conjunto de soluções reais da equação
  \[
    \frac{(x-1)^2}{4} + \frac{(y+2)^2}{9} = 0
  \]

\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
  Cada parcela é não negativa
  \[
    \pause
    \frac{(x-1)^2}{4}\geq 0
    \qquad\qquad
    \pause
    \frac{(y+2)^2}{9}\geq 0
  \]
  \pause
  A soma é nula somente se
  \[
    \pause
    x - 1 = 0
    \pause
    \qquad\text{e}\qquad
    y + 2 = 0
  \]
  \pause
  Conjunto solução
  \[
    S = \{(1,-2)\}
  \]
\end{resolution}
