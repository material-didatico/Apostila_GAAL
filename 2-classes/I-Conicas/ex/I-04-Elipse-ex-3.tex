\begin{question}
  Determine o conjunto de soluções reais da equação
  \[
    \frac{(x-1)^2}{4} + \frac{(y+2)^2}{9} = 0
  \]

\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
  Cada parcela é não negativa
  \[
    \frac{(x-1)^2}{4}\geq 0
    \qquad
    \frac{(y+2)^2}{9}\geq 0
  \]
  A soma é nula somente se
  \[
    x - 1 = 0
    \qquad
    y + 2 = 0
  \]
  Logo o conjunto de soluções é
  \[
    S = \{(1,-2)\}
  \]
\end{resolution}
