\begin{question}
  Considerando a equação
  \[
    9x^2 + 4y^2 = 36
  \]
  \vspace{-\baselineskip}
  \begin{enumerate}
    \item identifique a curva
    \item determine seus principais valores e pontos
    \item esboce a curva
  \end{enumerate}
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Identificação}
\begin{columns}
\column{0.4\textwidth}
  \begin{align*}
    \onslide<+->{9x^2 + 4y^2                       & = 36 \\}
    \onslide<+->{\frac{9x^2}{36} + \frac{4y^2}{36} & = 1  \\}
    \onslide<+->{\frac{x^2}{4}   + \frac{y^2}{9}   & = 1}
  \end{align*}
  \onslide<+->{comparando com
   \[
     \frac{x^2}{a^2} + \frac{y^2}{b^2} = 1
   \]}
\column{0.4\textwidth}
  \onslide<+->{A curva é uma elipse}

  \bigskip
  \onslide<+->{Centro: \(C = (0, 0)\)}

  \[
    \onslide<+->{a^2 = 4}
    \qquad
    \onslide<+->{a = 2}
  \]

  \[
    \onslide<+->{b^2 = 9}
    \qquad
    \onslide<+->{b = 3}
  \]
\end{columns}
\end{resolution}
