\begin{question}
  Dada a equação da circunferência
  \[
    x^2 + y^2 = 16
  \]
  \vspace{-1.5\baselineskip}
  \begin{enumerate}
    \item encontre seu raio e o centro
    \item escreva a equação matricial
    \item esboce a curva
  \end{enumerate}
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Raio de Centro}
  Comparando
  \[
    x^2 + y^2 = 16
  \]
  \pause
  com a equação canônica
  \[
    x^2 + y^2 = r^2
  \]
  \pause
  Identificamos

  \pause
  \quad o centro:
  \(
    C = (0, 0)
  \)

  \pause
  \quad o raio:
  \(
    r = \sqrt{16} = 4
  \)
\end{resolution}

%------------------------------------------------------------------------------%
\begin{resolution}{Equação Matricial}
  Equação Matricial
  \pause
  \[
    v^t I v = r^2
  \]
  \pause
  com
  \[
    I = \begin{bmatrix}
      1 & 0 \\
      0 & 1
    \end{bmatrix}
    \qquad\qquad
    \pause
    r = 4
  \]
\end{resolution}
