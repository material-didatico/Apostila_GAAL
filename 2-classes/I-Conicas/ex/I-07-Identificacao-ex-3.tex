\begin{question}
  Considerando a equação
  \[
    4x^2 - 9y^2 = 36
  \]
  \vspace{-\baselineskip}
  \begin{enumerate}
    \item identifique a curva
    \item determine seus principais valores e pontos
    \item esboce a curva
  \end{enumerate}
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Identificação}
\begin{columns}
\column{0.4\textwidth}
  \begin{align*}
    \onslide<+->{4x^2            - 9y^2            & = 36 \\}
    \onslide<+->{\frac{4x^2}{36} - \frac{9y^2}{36} & = 1  \\}
    \onslide<+->{\frac{x^2}{9}   - \frac{y^2}{4}   & = 1}
  \end{align*}
  \onslide<+->{comparando com
   \[
     \frac{x^2}{a^2} - \frac{y^2}{b^2} = 1
   \]}
   \onslide<+->{A curva é uma hipérbole}
\column{0.4\textwidth}
  \onslide<+->{Centro: \(C = (0, 0)\)}

  \[
    \onslide<+->{a^2 = 9}
    \qquad
    \onslide<+->{a = 3}
  \]

  \[
    \onslide<+->{b^2 = 4}
    \qquad
    \onslide<+->{b = 2}
  \]
  \onslide<+->{Como o termo positivo é o $x^2$}

  \onslide<+->{os vértices estão no eixo $x$}

  \bigskip
  \onslide<+->{Vértices: $(-3, 0)$ e $(3, 0)$}
\end{columns}
\end{resolution}
