\begin{question}
  Considerando a equação
  \[
    y^2 = 12x
  \]
  \vspace{-\baselineskip}
  \begin{enumerate}
    \item identifique a curva
    \item determine seus principais valores e pontos
    \item esboce a curva
  \end{enumerate}
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Identificação}
  Comparando
  \tabto{30mm}
  \(
    y^2 = 12x
  \)

  \pause
  com
  \tabto{30mm}
  \(
    y^2 = 4px
  \)

  \pause
  A curva é uma parábola
  \pause
  com vértice em $(0, 0)$
  \[
    \pause
    4p = 12
    \qquad
    \qquad
    \pause
    p = 3
  \]
  \pause
  O termo elevado ao quadrado é $y^2$

  \pause
  O eixo de simetria é paralelo ao eixo $x$

  \pause
  Como $p > 0$ a parábola tem abertura para a direita
\end{resolution}
