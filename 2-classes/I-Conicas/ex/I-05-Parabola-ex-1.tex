\begin{question}
  Dada a equação da parábola
  \[
    x^2 = 8y
  \]
  \vspace{-1.5\baselineskip}
  \begin{enumerate}
    \item identifique o parâmetro $p$
    \item escreva a equação matricial
    \item esboce a curva
  \end{enumerate}
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Parâmetro $p$}
  Comparando
  \[
    x^2 = 8y
  \]
  \pause
  com a equação canônica
  \[
    x^2 = 4py
  \]
  \pause
  identificamos
  \[
    \pause
    4p = 8
    \qquad
    \pause
    p = 2
  \]
\end{resolution}

%------------------------------------------------------------------------------%
\begin{resolution}{Equação Matricial}
  Equação Matricial
  \[
    v^tAv + 2B^tv = 0
  \]
  \pause
  com
  \[
    \pause
    A = \begin{bmatrix}
      1 & 0 \\
      0 & 0
    \end{bmatrix}
    \qquad\qquad
    \pause
    B
    = \begin{bmatrix}[c]
      0   \\
      -2p
    \end{bmatrix}
    \pause
    = \begin{bmatrix}
      0  \\
      -4
    \end{bmatrix}
  \]
\end{resolution}
