\begin{question}
  Determine o conjunto das soluções reais de
  \[
    (x-1)^2 + (y+2)^2 = 0
  \]
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
  Como os quadrados são não negativos
  \[
    (x-1)^2 \geq 0
    \qquad
    (y+2)^2 \geq 0
  \]
  A soma só é nula se
  \[
    x - 1 = 0
    \qquad
    y + 2 = 0
  \]
  Logo o conjunto de soluções é
  \[
    S = \{(1, -2)\}
  \]
  A equação representa uma circunferência degenerada em um ponto
\end{resolution}
