\begin{question}
  Determine e esboce o conjunto das soluções reais de
  \[
    (x-1)^2 + (y+2)^2 = 0
  \]
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
  Como os quadrados são não negativos
  \[
    \pause
    (x-1)^2 \geq 0
    \qquad\qquad
    \pause
    (y+2)^2 \geq 0
  \]
  \pause
  A soma só é nula se
  \[
    \pause
    x - 1 = 0
    \pause
    \qquad\text{e}\qquad
    y + 2 = 0
  \]
  \pause
  Logo o conjunto de soluções é
  \[
    \pause
    S = \{(1, -2)\}
  \]
  \pause
  A equação representa uma circunferência degenerada em um ponto
\end{resolution}
