\begin{question}
  Determine e esboce o conjunto de soluções reais da equação
  \[
    x^2 - y^2 = 0
  \]
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
  \begin{align*}
    \onslide<+->{x^2 - y^2  & = 0                              \\}
    \onslide<+->{(x-y)(x+y) & = 0                              \\}
    \onslide<+->{x-y        & = 0}
    \onslide<+->{\qquad\text{ou}\qquad x+y=0}
  \end{align*}

  \onslide<+->{O conjunto de soluções é a união das retas}
  \onslide<+->{\(
    y = x
  \)}
  \onslide<+->{ e
  \(
    y = -x
  \)}
  \onslide<+->{\[
    S = \left\{ (x, y) \in \R^2 \st y = x \;\text{ ou }\; y = -x \right\}
  \]}
  \onslide<+->{Essa equação é um caso degenerado da equação quadrática}
\end{resolution}
