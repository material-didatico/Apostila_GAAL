\begin{frame}{Equações Canônicas}
  Circunferência
  \tabto{35mm}
  \(
    x^2 + y^2 = r^2
  \)
  \tabto{90mm}
  \(
    r > 0
  \)

  \pause
  \bigskip
  \bigskip
  Elipse
  \tabto{35mm}
  \(
    \frac{x^2}{a^2} + \frac{y^2}{b^2} = 1
  \)
  \tabto{90mm}
  \(
    a > 0
    \qquad
    b > 0
  \)

  \pause
  \bigskip
  \bigskip
  Parábola
  \tabto{35mm}
  \(
    x^2 = 4py
  \)
  \tabto{90mm}
  \(
    p \neq 0
  \)

  \pause
  \bigskip
  \bigskip
  Hipérbole
  \tabto{35mm}
  \(
    \frac{x^2}{a^2} - \frac{y^2}{b^2} = 1
  \)
  \tabto{90mm}
  \(
    a > 0
    \qquad
    b > 0
  \)
\end{frame}

%------------------------------------------------------------------------------%
\begin{frame}{Equações Canônicas Transladadas}

    a > 0
    \qquad
    b > 0
  \)
\end{frame}
