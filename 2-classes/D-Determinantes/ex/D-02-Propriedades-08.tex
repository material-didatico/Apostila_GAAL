\begin{question}
  Determine os valores de $k$ para os quais a matriz
  \[
    A =
    \begin{pmatrix}
      1 & 2 & 0 \\
      0 & k & 1 \\
      0 & 0 & 3
    \end{pmatrix}
  \]
  é invertível
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
  \onslide<+->{Como $A$ é triangular}
  \[
    \onslide<+->{\det(A)}
    \onslide<+->{\;=\; a_{11} \; a_{22} \; a_{33}}
    \onslide<+->{\;=\; 1 \times k \times 3}
    \onslide<+->{\;=\; 3k}
  \]

  \onslide<+->{A matriz é invertível quando}
  \begin{align*}
    \onslide<+->{\det(A) & \neq 0 \\}
    \onslide<+->{3k      & \neq 0 \\}
    \onslide<+->{k       & \neq 0   }
  \end{align*}
\end{resolution}
