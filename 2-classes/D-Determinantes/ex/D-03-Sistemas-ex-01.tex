\begin{question}
  Utilize o determinantes para verificar se o sistema possui solução única
  \[
    \begin{cases}
      2x + y = 5 \\
       x - y = 1
    \end{cases}
  \]
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
  Matriz dos coeficientes
  \[
    A =
    \begin{pmatrix}
      2 & 1  \\
      1 & -1
    \end{pmatrix}
  \]

  \pause
  Determinante
  \[
    \det(A)
    \pause
    \;=\; 2(-1) - 1(1)
    \pause
    \;=\; -3
    \pause
    \;\neq\; 0
  \]
  \pause
  o sistema possui uma única solução
\end{resolution}
