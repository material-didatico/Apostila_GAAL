\begin{question}
  Calcule o determinante da matriz
  \[
    A =
    \begin{pmatrix}
      2 & 3 & 1  \\
      0 & 4 & 5  \\
      0 & 0 & -2
    \end{pmatrix}
  \]
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
  \onslide<+->{\[
    A =
    \begin{pmatrix}
      \emph{2} & 3        & 1         \\
      \z       & \emph{4} & 5         \\
      \z       & \z       & \emph{-2}
    \end{pmatrix}
  \]}

  \begin{align*}
    \onslide<+->{\det(A)}
    \onslide<+->{& = a_{11} \; a_{22} \; a_{22} \\}
    \onslide<+->{& = 2 \times 4 \times (-2) \\}
    \onslide<+->{& = -16}
  \end{align*}
\end{resolution}
