\begin{question}
  Calcule o determinante da matriz
  \[
    A =
    \begin{pmatrix}
      1 & 2 & 3 \\
      2 & 4 & 6 \\
      1 & 0 & 1
    \end{pmatrix}
  \]
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
  A segunda linha é duas vezes a primeira
  \[
    \det(A) = 0
  \]
\end{resolution}
