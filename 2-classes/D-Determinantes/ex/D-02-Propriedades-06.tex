\begin{question}
  Verifique se a matriz é invertível
  \[
    A =
    \begin{pmatrix}
      1 & 2 & 3 \\
      0 & 1 & 4 \\
      0 & 0 & 2
    \end{pmatrix}
  \]
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
  A matriz é triangular
  \pause
  \[
    \det(A)
    \pause
    = a_{11} \; a_{22} \; a_{33}
    \pause
    = 1 \times 1 \times 2
    \pause
    = 2
  \]

  \pause
  Como
  \[
    \det(A) \neq 0
  \]
  \pause
  \(A\) é invertível
\end{resolution}
