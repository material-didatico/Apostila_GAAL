\begin{question}
  Verifique se a matriz é invertível
  \[
    B =
    \begin{pmatrix}
      1 & 2 & 3 \\
      2 & 4 & 6 \\
      0 & 1 & 2
    \end{pmatrix}
  \]
\end{question}

%------------------------------------------------------------------------------%
\begin{resolution}{Solução}
  A segunda linha é o dobro da primeira

  \pause
  Portanto
  \[
    \det(B) = 0
  \]

  \pause
  Assim
  \(B\) não é invertível
\end{resolution}
